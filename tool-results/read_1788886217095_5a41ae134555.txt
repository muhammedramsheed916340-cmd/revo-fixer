    74→let cachedSignals: Prediction[] | null | undefined;
    75→let cachedRaw = "";
    76→
    77→function readSavedSignals(): Prediction[] | null {
    78→  if (typeof window === "undefined") return null;
    79→  try {
    80→    const saved = localStorage.getItem(SIGNALS_KEY) ?? "";
    81→    if (saved === cachedRaw && cachedSignals !== undefined) {
    82→      return cachedSignals;
    83→    }
    84→    cachedRaw = saved;
    85→    if (!saved) {
    86→      cachedSignals = null;
    87→      return null;
    88→    }
    89→    const data = JSON.parse(saved) as Prediction[];
    90→    if (!Array.isArray(data) || data.length === 0) {
    91→      localStorage.removeItem(SIGNALS_KEY);
    92→      cachedSignals = null;
    93→      return null;
    94→    }
    95→    if (Date.now() - data[0].time > 5 * 60 * 1000) {
    96→      localStorage.removeItem(SIGNALS_KEY);
    97→      cachedSignals = null;
    98→      return null;
    99→    }
   100→    cachedSignals = data;
   101→    return data;
   102→  } catch {
   103→    cachedSignals = null;
   104→    return null;
   105→  }
   106→}
   107→
   108→function subscribeSignals(cb: () => void): () => void {
   109→  signalsListeners.add(cb);
   110→  const onStorage = (e: StorageEvent) => {
   111→    if (e.key === SIGNALS_KEY) cb();
   112→  };
   113→  window.addEventListener("storage", onStorage);
   114→  return () => {
   115→    signalsListeners.delete(cb);
   116→    window.removeEventListener("storage", onStorage);
   117→  };
   118→}
   119→

   121→  return useSyncExternalStore(
   122→    subscribeSignals,
   123→    readSavedSignals,
   124→    () => null,

   169→function readRoundHistory(): RoundResult[] {
   170→  if (typeof window === "undefined") return EMPTY_ROUNDS;
   171→  try {
   172→    const saved = localStorage.getItem(ROUNDS_KEY) ?? "";
   173→    if (saved === cachedRoundsRaw && cachedRounds !== undefined) {
   174→      return cachedRounds;
   175→    }
   176→    cachedRoundsRaw = saved;
   177→    if (!saved) {
   178→      cachedRounds = EMPTY_ROUNDS;
   179→      return EMPTY_ROUNDS;
   180→    }
   181→    const data = JSON.parse(saved) as StoredRound[];
   182→    if (!Array.isArray(data)) {
   183→      cachedRounds = EMPTY_ROUNDS;
   184→      return EMPTY_ROUNDS;
   185→    }
   186→    const rounds: RoundResult[] = data.map((r) => ({
   187→      prediction: (r.prediction ?? []).map((p) => ({
   188→        game: GAMES.find((g) => g.name === p.game.name) ?? GAMES[0],
   189→        confidence: p.confidence,
   190→        time: p.time,
   191→      })),
   192→      actualResult: GAMES.find((g) => g.name === r.actualResult.name) ?? GAMES[0],
   193→      hit: r.hit,
   194→      time: r.time,
   195→      confidence: r.confidence ?? 0,
   196→      recalibrated: r.recalibrated ?? false,
   197→      calibrationNote: r.calibrationNote,
   198→    }));
   199→    cachedRounds = rounds;
   200→    return rounds;
   201→  } catch {
   202→    cachedRounds = EMPTY_ROUNDS;
   203→    return EMPTY_ROUNDS;
   204→  }
   205→}

   260→// ============================================================
   261→// SHADOW A/B LEDGER — baseline vs experimental on the SAME rounds
   262→// ============================================================
   263→// Each live round is settled TWICE in parallel:
   264→//   - baseline prediction (frozen k=30, the active displayed prediction)
   265→//   - experimental prediction (k=30 + rare-outcome reliability layer)
   266→// Both are computed from the SAME pre-result history → no data leakage.
   267→// The ledger persists across reloads so a 50-round A/B accumulates live.
   268→const SHADOW_KEY = "revo_shadowLedger";
   269→const EXP_FLAG_KEY = "revo_experimentalFlag";

   271→interface ShadowRow {
   272→  ts: number;
   273→  actual: string;
   274→  baselinePreds: string[];
   275→  baselineHit: boolean;
   276→  expPreds: string[];
   277→  expHit: boolean;

   284→function readShadowLedger(): ShadowRow[] {
   285→  if (typeof window === "undefined") return EMPTY_SHADOW;

   287→    const raw = localStorage.getItem(SHADOW_KEY) ?? "";
   288→    if (raw === cachedShadowRaw && cachedShadow !== undefined) {
   289→      return cachedShadow;

   297→    cachedShadow = Array.isArray(data) ? (data as ShadowRow[]) : EMPTY_SHADOW;
   298→    return cachedShadow;
   299→  } catch {
   300→    cachedShadow = EMPTY_SHADOW;
   301→    return EMPTY_SHADOW;

   401→// ============================================================
   402→// MAIN COMPONENT
   403→// ============================================================
   404→export function RevoGame() {
   405→  const savedSignals = useSavedSignals();
   406→  const roundHistory = useRoundHistory();
   407→  const [predictions, setPredictions] = useState<Prediction[] | null>(null);
   408→  const [loading, setLoading] = useState(false);
   409→  const [running, setRunning] = useState(false);
   410→  const [lastRecalibration, setLastRecalibration] = useState<{ triggered: boolean; reason: string } | null>(null);
   411→  // NOTE: NO countdown, NO auto-refresh. Prediction is LOCKED until the next
   412→  // live result arrives (per user spec — LIVE AUTO mode, no manual intervention).
   413→  // Hydration guard — `savedSignals`/`roundHistory` come from localStorage which
   414→  // is null on the server but non-null on the client after mount. Rendering
   415→  // prediction-derived UI before mount causes hydration mismatches.
   416→  // Uses useSyncExternalStore (server snapshot = false, client = true after subscribe)
   417→  // which is the lint-safe pattern for mount detection.
   418→  const mounted = useSyncExternalStore(
   419→    () => () => {},
   420→    () => true,
   421→    () => false,
   422→  );
   423→  const [stats, setStats] = useState({
   424→    total: 1249,
   425→    accuracy: 94,
   426→    bonusHits: 128,
   427→    liveUsers: 1247,
   428→  });
   429→  const [clock, setClock] = useState("");
   430→  const liveRef = useRef<ReturnType<typeof setInterval> | null>(null);
   431→
   432→  // Gate prediction-derived UI by `mounted` to avoid SSR hydration mismatch
   433→  // (localStorage is null on server but non-null on client after mount).
   434→  const displayPredictions = mounted ? (predictions ?? savedSignals) : null;
   435→  const isRunning = running || (predictions === null && savedSignals !== null && !loading);
   436→
   437→  // ===== REAL CASINO SPINS (from CasinoScores API, shared via liveSpinStore) =====
   438→  // RevoLiveResults writes the latest 30 real spins here. We subscribe so the
   439→  // engine re-derives predictions whenever fresh real casino data arrives.
   440→  const liveSpins = useSyncExternalStore(
   441→    subscribeLiveSpins,
   442→    getLiveSpins,
   443→    () => EMPTY_SPINS, // server snapshot — stable empty constant
   444→  );
   445→
   446→  // ===== UNIFIED ENGINE OUTPUT =====
   447→  // This is THE single source of truth for ALL UI sections.
   448→  // Re-computed whenever round history OR live casino spins change.
   449→  // Passes REAL casino spins to the engine so predictions are data-driven
   450→  // (not always [1,2,5,10]) and varied via weighted probabilistic sampling.
   451→  const engine: EngineOutput = useMemo(() => {
   452→    return buildInitial(roundHistory, liveSpins);
   453→  }, [roundHistory, liveSpins]);
   454→
   455→  // ===== STABLE PREDICTION VIEW =====
   456→  // The engine re-samples every time liveSpins changes (every 4s poll). But
   457→  // per the stability rule, the prediction cards must NOT change until the
   458→  // next live result arrives. So we derive the prediction-derived fields
   459→  // (predictions, excludedOutcomes, nextSignalNames, excludedNames) from the
   460→  // STABLE displayPredictions (state/localStorage), while using the engine's
   461→  // live analysis (dashboard, candidateScores, RCA, confidence, decision).
   462→  const view: EngineOutput = useMemo(() => {
   463→    const active = displayPredictions ?? engine.predictions.map((p) => ({
   464→      game: p.game,
   465→      confidence: p.confidence,
   466→      time: p.time,
   467→    }));
   468→    const activeNames = active.map((p) => p.game.name);
   469→    const excludedGames = ENGINE_GAMES.filter((g) => !activeNames.includes(g.name));
   470→    return {
   471→      ...engine,
   472→      // Stable prediction-derived fields (from displayPredictions):
   473→      nextSignalNames: activeNames,
   474→      excludedOutcomes: excludedGames,
   475→      excludedNames: excludedGames.map((g) => g.name),
   476→      // Keep engine's analysis fields (dashboard, candidateScores, RCA, etc.)
   477→    };
   478→  }, [engine, displayPredictions]);
   479→
   480→  // Derived display values (hydration-safe)
   481→  const verifiedRounds = roundHistory.length;
   482→  const hits = roundHistory.filter((r) => r.hit).length;
   483→  const realAccuracy =
   484→    verifiedRounds > 0 ? Math.round((hits / verifiedRounds) * 100) : null;
   485→
   486→  // Live clock (HH:MM)
   487→  useEffect(() => {
   488→    const tick = () => {
   489→      const d = new Date();
   490→      setClock(
   491→        `${String(d.getHours()).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")}`,
   492→      );
   493→    };
   494→    tick();
   495→    const t = setInterval(tick, 1000);
   496→    return () => clearInterval(t);
   497→  }, []);
   498→
   499→  // ===== GENERATE PREDICTION (initial only — NO auto-refresh) =====
   500→  // Prediction is LOCKED once generated. It only changes when a new LIVE
   501→  // result arrives (via selectActualResult). No countdown, no refresh,
   502→  // no manual intervention — pure LIVE AUTO mode.
   503→  const generatePrediction = useCallback(() => {
   504→    setLoading(true);
   505→    setPredictions(null);
   506→    const allRounds = readRoundHistory();
   507→    const spins = getLiveSpins();
   508→    const eng = buildInitial(allRounds, spins);
   509→    const preds = engineToPredictions(eng);
   510→    setPredictions(preds);
   511→    setLoading(false);
   512→    setRunning(true);
   513→    saveSignals(preds);
   514→    setStats((s) => {
   515→      const total = s.total + preds.length;
   516→      const bonusHits =
   517→        s.bonusHits +
   518→        preds.filter((p) => BONUS_NAMES.includes(p.game.name)).length;
   519→      return { ...s, total, bonusHits, accuracy: s.accuracy };
   520→    });
   521→    // SHADOW: seed the experimental engine's locked prediction (same history).
   522→    const expEng = buildInitial(allRounds, spins, "experimental");
   523→    expLockedNames = expEng.predictions.map((p) => p.game.name);
   524→  }, []);
   525→
   526→  // AUTO-GENERATE on mount ONLY (no countdown refresh)
   527→  const autoStarted = useRef(false);
   528→  useEffect(() => {
   529→    if (autoStarted.current) return;
   530→    if (savedSignals) return;
   531→    autoStarted.current = true;
   532→    const t = setTimeout(() => generatePrediction(), 100);
   533→    return () => clearTimeout(t);
   534→  }, [savedSignals, generatePrediction]);
   535→
   536→  // Event debug log — proves the exact pipeline order for each live result
   537→  const [eventLog, setEventLog] = useState<Array<{
   538→    eventId: number;
   539→    actual: string;
   540→    oldPrediction: string[];
   541→    hitMiss: string;
   542→    historyNBefore: number;
   543→    historyNAfter: number;
   544→    newPrediction: string[];
   545→    predictionId: number;
   546→    locked: boolean;
   547→    timestamp: number;
   548→  }>>([]);
   549→
   550→  // ===== SHADOW A/B: baseline vs experimental on the SAME live rounds =====
   551→  // experimentalEnabled is a FEATURE FLAG. When ON, the experimental engine
   552→  // (k=30 + rare-outcome reliability layer) ALSO runs in shadow — its
   553→  // prediction is locked at the same moment as baseline, settled against
   554→  // the same actual, and recorded in `shadowLedger`. The ACTIVE displayed
   555→  // prediction remains the frozen k=30 baseline (unchanged behavior).
   556→  const experimentalEnabled = useSyncExternalStore(
   557→    subscribeExpFlag,
   558→    readExpFlag,
   559→    () => false, // server snapshot
   560→  );
   561→  const shadowLedger = useSyncExternalStore(
   562→    subscribeShadow,
   563→    readShadowLedger,
   564→    () => EMPTY_SHADOW, // server snapshot — stable constant
   565→  );
   566→  // Retrospective diagnostic result (computed on demand).
