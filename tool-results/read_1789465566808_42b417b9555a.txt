     8→function timeAgo(ts: string): string {
     9→  const diff = Date.now() - new Date(ts).getTime();
    10→  if (diff < 60000) return `${Math.floor(diff / 1000)}s ago`;
    11→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`;
    12→  return `${Math.floor(diff / 3600000)}h ago`;
    13→}
