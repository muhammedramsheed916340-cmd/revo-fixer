     9→function timeAgo(ts: string): string {
    10→  const diff = Date.now() - new Date(ts).getTime();
    11→  if (diff < 60000) return `${Math.floor(diff / 1000)}s ago`;
    12→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`;
    13→  return `${Math.floor(diff / 3600000)}h ago`;
    14→}

   111→              actualOutcome,
   112→              actualSectorIdx >= 0 ? actualSectorIdx : null,

   114→          }
   115→        }
