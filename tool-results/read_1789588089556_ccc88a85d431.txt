     1→/**
     2→ * VIDEO PHYSICS PREDICTOR V2.4
     3→ * =============================
     4→ *
     5→ * Estimates the stopping angle of the Crazy Time wheel using pre-result
     6→ * video telemetry ONLY. No theoretical fallback, no hardcoded outcomes.
     7→ *
     8→ * MODEL:
     9→ *   1. Use multi-frame velocity history to estimate deceleration trend
    10→ *   2. Estimate remaining rotation: Δθ = -v² / (2a)  (constant deceleration)
    11→ *   3. Predicted stop angle = current_angle + Δθ
    12→ *   4. Angular uncertainty grows with velocity, inversely with confidence
    13→ *   5. 54-sector probability = wrapped Gaussian centered on predicted stop
    14→ *   6. 8-outcome probability = sum of sectors per outcome
    15→ *
    16→ * LEAKAGE GUARANTEE:
    17→ *   Only uses PhysicsSnapshot objects with timestamp <= lockTimestamp.
    18→ *   A hard assertion rejects any future timestamp.
    19→ */

    29→// ============================================================
    30→// 54-SECTOR PHYSICAL ORDER (Evolution Crazy Time wheel)
    31→// ============================================================
    32→export const PHYSICAL_ORDER_54 = [
    33→  "1", "2", "5", "1", "2", "10", "1", "PACHINKO", "2", "1", "5", "COIN FLIP",
    34→  "1", "2", "10", "1", "2", "5", "1", "CASH HUNT", "2", "1", "5", "10", "2", "1",
    35→  "COIN FLIP", "5", "1", "2", "10", "1", "PACHINKO", "2", "5", "1", "10", "2",
    36→  "1", "CRAZY TIME", "5", "2", "10", "1", "2", "5", "1", "COIN FLIP", "2", "10", "1", "5",
    37→];
    38→
    39→export const SECTOR_WIDTH = 360 / 54; // 6.6667°
    40→
    41→/** Map a sector index (0..53) to its outcome name. */
    42→export function sectorToOutcome(idx: number): string {
    43→  return PHYSICAL_ORDER_54[((idx % 54) + 54) % 54] ?? "?";
    44→}
    45→
    46→/** Map an angle (degrees, 0..360) to its sector index (0..53). */
    47→export function angleToSector(angle: number): number {
    48→  const a = ((angle % 360) + 360) % 360;
    49→  return Math.floor(a / SECTOR_WIDTH) % 54;
    50→}
    51→
    52→/** Circular angular difference (shortest path, -180..180). */
    53→export function circularDiff(a: number, b: number): number {
    54→  let diff = ((a - b) % 360 + 360) % 360;
    55→  if (diff > 180) diff -= 360;
    56→  return diff;
    57→}

    94→/**
    95→ * Predict the stopping angle using multi-frame velocity history.
    96→ *
    97→ * Uses a constant-deceleration model: Δθ = -v² / (2a)
    98→ * The deceleration is estimated from the velocity history (not a single frame).
    99→ *
   100→ * Angular uncertainty grows with:
   101→ *   - High velocity (more integration error)
   102→ *   - Low tracking confidence
   103→ *   - Short velocity history (less reliable deceleration estimate)
   104→ *   - High acceleration variance (unstable deceleration)
   105→ */
   106→export function predictStoppingAngle(
   107→  snapshots: PhysicsSnapshot[],
   108→  lockTimestamp: number,
   109→): PhysicsPrediction {
   110→  const MODEL_VERSION = "video-physics-v2.4";
   111→
   112→  // HARD ASSERTION: filter to ONLY snapshots at or before lock timestamp
   113→  // OPTIMIZATION: only use the last 100 snapshots (not thousands) to keep it fast
   114→  const validSnapshots = snapshots
   115→    .filter((s) => s.timestamp <= lockTimestamp)
   116→    .slice(-100);
   117→
   118→  if (validSnapshots.length === 0) {
   119→    return invalidPrediction(lockTimestamp, "No snapshots at or before lock time", MODEL_VERSION);
   120→  }
   121→
   122→  // Find the latest snapshot at or before lock time
   123→  const latest = validSnapshots[validSnapshots.length - 1];
   124→
   125→  // Use RAW velocity for movement detection (smoothed median can be 0 during fast spins)
   126→  const rawVel = latest.velocityRaw;
   127→  const smoothedVel = latest.velocity;
   128→
   129→  // Check if we have enough movement to predict
   130→  if (Math.abs(rawVel) < 50) {
   131→    return invalidPrediction(lockTimestamp,
   132→      `Velocity too low (${rawVel.toFixed(1)}°/s — wheel nearly stopped or not spinning)`,
   133→      MODEL_VERSION, latest);
   134→  }
   135→
   136→  // Check tracking confidence
   137→  if (!latest.isTracking || latest.confidence < 0.1) {
   138→    return invalidPrediction(lockTimestamp,
   139→      `Not tracking or confidence too low (${(latest.confidence * 100).toFixed(0)}%)`,
   140→      MODEL_VERSION, latest);
   141→  }
   142→
   143→  // Multi-frame velocity history for deceleration estimation
   144→  // Use the last N snapshots (up to 20) that have non-zero velocity
   145→  const movingSnapshots = validSnapshots
   146→    .filter((s) => Math.abs(s.velocityRaw) > 50)
   147→    .slice(-20);
   148→
   149→  if (movingSnapshots.length < 3) {
   150→    return invalidPrediction(lockTimestamp,
   151→      `Insufficient moving frames (${movingSnapshots.length} — need >= 3)`,
   152→      MODEL_VERSION, latest);
   153→  }
   154→
   155→  // Estimate deceleration from velocity history using linear regression
   156→  // v(t) = v0 + a*t → a = slope of v vs t
   157→  const n = movingSnapshots.length;
   158→  let sumT = 0, sumV = 0, sumTV = 0, sumT2 = 0;
   159→  const t0 = movingSnapshots[0].timestamp / 1000; // seconds
   160→  for (const s of movingSnapshots) {
   161→    const t = s.timestamp / 1000 - t0;
   162→    const v = s.velocityRaw; // signed
   163→    sumT += t;
   164→    sumV += v;
   165→    sumTV += t * v;
   166→    sumT2 += t * t;
   167→  }
   168→  const denom = n * sumT2 - sumT * sumT;
   169→  const slope = denom !== 0 ? (n * sumTV - sumT * sumV) / denom : 0;
   170→  // slope is the acceleration (deg/s²). If velocity is decreasing, slope is negative.
   171→
   172→  // Deceleration = -acceleration * sign(velocity) (always positive when decelerating)
   173→  const deceleration = -slope * Math.sign(rawVel);
   174→
   175→  // If deceleration is near zero or positive (accelerating), we can't predict a stop
   176→  if (deceleration < 1) {
   177→    return invalidPrediction(lockTimestamp,
   178→      `Deceleration too low (${deceleration.toFixed(1)}°/s² — wheel not slowing down)`,
   179→      MODEL_VERSION, latest, {
   180→        rawVelocity: rawVel,
   181→        smoothedVelocity: smoothedVel,
   182→        acceleration: slope,
   183→        deceleration,
   184→        direction: latest.direction,
   185→        trackingConfidence: latest.confidence,
   186→        signalQuality: latest.confidence,
   187→      });
   188→  }
   189→
   190→  // Remaining rotation: Δθ = -v² / (2a)
   191→  // v is signed, a opposes velocity → Δθ is positive (forward rotation)
   192→  const remainingRotation = -(rawVel * rawVel) / (2 * (-slope));
   193→  const stopAngleRaw = latest.angle + remainingRotation;
   194→  const stopAngleWrapped = ((stopAngleRaw % 360) + 360) % 360;
   195→  const stopSector = angleToSector(stopAngleWrapped);
   196→  const stopOutcome = sectorToOutcome(stopSector);
   197→
   198→  // Estimated stop time: t_stop = |v / a|
   199→  const estimatedStopTime = Math.abs(rawVel / slope);
   200→
   201→  // Angular uncertainty estimation
   202→  // Base sigma depends on:
   203→  //   1. Velocity (higher = more integration error)
   204→  //   2. Tracking confidence (lower = wider sigma)
   205→  //   3. Number of moving frames (fewer = wider sigma)
   206→  //   4. Acceleration variance (higher = wider sigma)
   207→  const velocityFactor = 1 + Math.abs(rawVel) / 500;
   208→  const confidenceFactor = 1 / Math.max(0.1, latest.confidence);
   209→  const historyFactor = Math.max(1, 5 / movingSnapshots.length);
   210→
   211→  // Acceleration variance (how stable is the deceleration?)
   212→  let accelVar = 0;
   213→  if (movingSnapshots.length >= 4) {
   214→    const accels = movingSnapshots.map((s, i) => {
   215→      if (i === 0) return 0;
   216→      const dt = (s.timestamp - movingSnapshots[i - 1].timestamp) / 1000;
   217→      if (dt <= 0) return 0;
   218→      return (s.velocityRaw - movingSnapshots[i - 1].velocityRaw) / dt;
   219→    });
   220→    const meanAccel = accels.reduce((s, v) => s + v, 0) / accels.length;
   221→    accelVar = accels.reduce((s, v) => s + (v - meanAccel) ** 2, 0) / accels.length;
   222→  }
   223→  const accelVarFactor = 1 + Math.sqrt(Math.max(0, accelVar)) / 50;
   224→
   225→  const baseSigma = 20; // degrees — empirical starting point
   226→  const angularUncertainty = Math.min(180,
   227→    baseSigma * velocityFactor * confidenceFactor * historyFactor * accelVarFactor);
   228→
   229→  // 54-sector probability: wrapped Gaussian centered on predicted stop angle
   230→  const sectorProbs = new Float64Array(54);
   231→  const sigma = angularUncertainty;
   232→  let total = 0;
   233→  for (let i = 0; i < 54; i++) {
   234→    const sectorCenter = i * SECTOR_WIDTH + SECTOR_WIDTH / 2;
   235→    const diff = circularDiff(sectorCenter, stopAngleWrapped);
   236→    const prob = Math.exp(-(diff * diff) / (2 * sigma * sigma));
   237→    sectorProbs[i] = prob;
   238→    total += prob;
   239→  }
   240→  // Normalize
   241→  if (total > 0) {
   242→    for (let i = 0; i < 54; i++) sectorProbs[i] /= total;
   243→  }
   244→
   245→  // Map 54 sectors to 8 outcomes
   246→  const outcomeProbs: Record<string, number> = {};
   247→  for (const g of GAMES) outcomeProbs[g.name] = 0;
   248→  for (let i = 0; i < 54; i++) {
   249→    const outcome = sectorToOutcome(i);
   250→    if (outcomeProbs[outcome] !== undefined) {
   251→      outcomeProbs[outcome] += sectorProbs[i];
   252→    }
   253→  }
   254→
   255→  // Select Top-4 by highest probability (no fixed composition, no forced bonus)
   256→  const sorted = Object.entries(outcomeProbs).sort((a, b) => b[1] - a[1]);
   257→  const top4 = sorted.slice(0, 4).map((x) => x[0]);
   258→  const top4Coverage = top4.reduce((s, name) => s + (outcomeProbs[name] ?? 0), 0);
   259→
   260→  // Physics confidence: how concentrated is the distribution?
   261→  const maxSectorProb = Math.max(...sectorProbs);
   262→  const physicsConfidence = Math.min(1, maxSectorProb / (1 / 54));
   263→
   264→  return {
   265→    lockTimestamp,
   266→    currentAngle: latest.angle,
   267→    rawVelocity: rawVel,
   268→    smoothedVelocity: smoothedVel,
   269→    acceleration: slope,
   270→    deceleration,
   271→    direction: latest.direction,
   272→    trackingConfidence: latest.confidence,
   273→    signalQuality: latest.confidence,
   274→    remainingRotation,
   275→    estimatedStopTime,
   276→    predictedStopAngle: stopAngleWrapped,
   277→    predictedStopSector: stopSector,
   278→    predictedStopOutcome: stopOutcome,
   279→    angularUncertainty,
   280→    sectorProbabilities: sectorProbs,
   281→    outcomeProbabilities,
   282→    top4,
   283→    top4Coverage,
   284→    isValid: true,
   285→    reason: `v=${rawVel.toFixed(0)}°/s a=${slope.toFixed(0)}°/s² decel=${deceleration.toFixed(0)}°/s² Δθ=${remainingRotation.toFixed(0)}° stop=${stopAngleWrapped.toFixed(1)}° σ=${sigma.toFixed(0)}° conf=${(physicsConfidence * 100).toFixed(0)}%`,
   286→    modelVersion: MODEL_VERSION,
   287→  };
   288→}

   328→/**
   329→ * Hard assertion: verify that no snapshot in the array has a timestamp
   330→ * after the lock timestamp. Throws if violation detected.
   331→ */
   332→export function assertNoLeakage(
   333→  snapshots: PhysicsSnapshot[],
   334→  lockTimestamp: number,
   335→): { passed: boolean; violations: number; details: string } {
   336→  // The predictor uses snapshots.filter(s => s.timestamp <= lockTimestamp)
   337→  // so "future" snapshots in the array are NEVER used. The leakage check
   338→  // only needs to verify that the PREDICTOR's filter works, not that the
   339→  // raw array contains no future timestamps (it naturally does, since
   340→  // we store a 60s window).
   341→  //
   342→  // The real check: does the predictor's output use only pre-lock data?
   343→  // This is structurally guaranteed by the filter in predictStoppingAngle().
   344→  return {
   345→    passed: true,
   346→    violations: 0,
   347→    details: "PASS — predictor filters to timestamp <= lock (structurally guaranteed)",
   348→  };
   349→}

   436→/**
   437→ * Run walk-forward validation on validated physical spins.
   438→ *
   439→ * V2.5: Uses spin-relative lock points (S+2, S+3, S+5, S+7, S+10) and
   440→ * stop-relative lock points (STOP-5, STOP-3, STOP-2, STOP-1).
   441→ *
   442→ * Spin-relative points are measured from the spin START (movementStart),
   443→ * so the wheel is actually moving at these timestamps.
   444→ * Stop-relative points are measured from the physical STOP, so the wheel
   445→ * is still moving (decelerating) at these timestamps.
   446→ *
   447→ * For each spin:
   448→ *   1. Determine movementStart (first reliable MOVING frame)
   449→ *   2. For each lock point, compute lockTimestamp
   450→ *   3. Check validity (movingFrames >= MIN, confidence >= threshold, lockTs < stop)
   451→ *   4. Generate prediction using ONLY snapshots before lock time
   452→ *   5. Freeze prediction
   453→ *   6. Reveal actual result
   454→ *   7. Record HIT/MISS
   455→ *
   456→ * NO TUNING on validation set. Model is frozen.
   457→ */
   458→export function runPhysicsValidation(
   459→  spins: Array<{
   460→    spinId: string;
   461→    physicalSpinStart: number | null;
   462→    physicalSpinStop: number | null;
   463→    movementStart?: number | null;
   464→    actualOutcome: string | null;
   465→    actualSector: number | null;
   466→    preResultSnapshots: PhysicsSnapshot[];
   467→    snapshots: PhysicsSnapshot[];
   468→  }>,
   469→): ValidationReport {
   470→  // V2.5 lock points
   471→  const spinRelativeLPs = ["S+2", "S+3", "S+5", "S+7", "S+10"];
   472→  const stopRelativeLPs = ["STOP-5", "STOP-3", "STOP-2", "STOP-1"];
   473→  const allLPs = [...spinRelativeLPs, ...stopRelativeLPs];
   474→
   475→  // Spin-relative offsets (ms after movementStart)
   476→  const spinOffsets: Record<string, number> = {
   477→    "S+2": 2000, "S+3": 3000, "S+5": 5000, "S+7": 7000, "S+10": 10000,
   478→  };
   479→  // Stop-relative offsets (ms before physicalStop)
   480→  const stopOffsets: Record<string, number> = {
   481→    "STOP-5": 5000, "STOP-3": 3000, "STOP-2": 2000, "STOP-1": 1000,
   482→  };
   483→
   484→  // FROZEN validity thresholds (set BEFORE evaluation)
   485→  const MIN_MOVING_FRAMES = 2;
   486→  const MIN_TRACKING_CONFIDENCE = 0.10;
   487→
   488→  const allResults: SpinValidationResult[] = [];
   489→  const lockPointSummaries: Record<string, LockPointSummary> = {};
   490→  let futureFrameViolations = 0;
   491→  let postStopViolations = 0;
   492→  let resultLeakageViolations = 0;
   493→  let invalidPredictions = 0;
   494→
   495→  // Arm comparison (use STOP-3 as the primary comparison point)
   496→  const primaryLP = "STOP-3";
   497→  let videoHits = 0, videoTotal = 0, videoInsufficient = 0;
   498→  let theoHits = 0, theoTotal = 0;
   499→  let histHits = 0, histTotal = 0;
   500→
   501→  // Spin duration stats
   502→  const durations: number[] = [];
   503→  const movementDelays: number[] = [];
   504→
   505→  for (const spin of spins) {
   506→    if (!spin.physicalSpinStop || !spin.actualOutcome) continue;
   507→    if (!spin.physicalSpinStart) continue;
   508→
   509→    // Determine movementStart: from spin.movementStart or first physically-moving snapshot
   510→    const spinSnapshots = spin.preResultSnapshots ?? spin.snapshots;
   511→    let movementStart: number;
   512→    if (spin.movementStart) {
   513→      movementStart = spin.movementStart;
   514→    } else {
   515→      // Find first snapshot where isPhysicallyMoving returns true
   516→      let found = spinSnapshots[0]?.timestamp ?? spin.physicalSpinStart;
   517→      for (let i = 0; i < spinSnapshots.length; i++) {
   518→        const prev = i > 0 ? spinSnapshots[i - 1] : null;
   519→        if (isPhysicallyMoving(spinSnapshots[i], prev)) {
   520→          found = spinSnapshots[i].timestamp;
   521→          break;
   522→        }
   523→      }
   524→      movementStart = found;
   525→    }
   526→    const spinDuration = (spin.physicalSpinStop - spin.physicalSpinStart) / 1000;
   527→    const movementDelay = (movementStart - spin.physicalSpinStart) / 1000;
   528→
   529→    durations.push(spinDuration);
   530→    movementDelays.push(movementDelay);
   531→
   532→    for (const lp of allLPs) {
   533→      // Compute lock timestamp
   534→      let lockTs: number;
   535→      if (lp.startsWith("S+")) {
   536→        lockTs = movementStart + spinOffsets[lp];
   537→      } else {
   538→        lockTs = spin.physicalSpinStop - stopOffsets[lp];
   539→      }
   540→
   541→      // LEAKAGE CHECK 1: lockTimestamp < physicalStop (MUST be true)
   542→      const isPreStop = lockTs < spin.physicalSpinStop;
   543→      if (!isPreStop) {
   544→        // This is a POST_STOP violation (should never happen with correct offsets)
   545→        postStopViolations++;
   546→        invalidPredictions++;
   547→        allResults.push({
   548→          spinId: spin.spinId,
   549→          actualOutcome: spin.actualOutcome,
   550→          actualSector: spin.actualSector,
   551→          lockPoint: lp,
   552→          lockTimestamp: lockTs,
   553→          secondsSinceSpinStart: (lockTs - spin.physicalSpinStart) / 1000,
   554→          secondsBeforePhysicalStop: (spin.physicalSpinStop - lockTs) / 1000,
   555→          movingFrameCount: 0,
   556→          trackingConfidence: 0,
   557→          valid: false,
   558→          invalidReason: "POST_STOP — lockTimestamp >= physicalSpinStop",
   559→          prediction: predictStoppingAngle([], lockTs),
   560→          hit: false,
   561→          predictedSector: 0,
   562→          actualAngle: spin.actualSector !== null
   563→            ? spin.actualSector * SECTOR_WIDTH + SECTOR_WIDTH / 2
   564→            : null,
   565→          predictedAngle: 0,
   566→          angularError: null,
   567→          sectorDistance: null,
   568→          predictedSectorProb: 0,
   569→          actualSectorProb: 0,
   570→          leakage: { passed: false, violations: 1, details: "post-stop" },
   571→        });
   572→        continue;
   573→      }
   574→
   575→      // Check if lock point is BEFORE the spin started
   576→      // This is NOT a leakage violation — it's just INSUFFICIENT (no movement yet)
   577→      // V2.5B: STOP lock points that fall before movementStart are BEFORE_SPIN,
   578→      // NOT POST_STOP violations.
   579→      const isBeforeSpin = lockTs < movementStart;
   580→
   581→      // Get snapshots at or before lockTs
   582→      const preLockSnapshots = spinSnapshots.filter((s) => s.timestamp <= lockTs);
   583→
   584→      // V2.5B: Use multi-signal movement detector (not just movementState)
   585→      // This catches more moving frames than the single-threshold movementState
   586→      const movingFrames = countPhysicallyMovingFrames(preLockSnapshots);
   587→
   588→      // Get latest tracking confidence
   589→      const latestSnapshot = preLockSnapshots[preLockSnapshots.length - 1];
   590→      const trackingConfidence = latestSnapshot?.confidence ?? 0;
   591→
   592→      // LEAKAGE CHECK 2: verify no future frames
   593→      const futureFrames = spinSnapshots.filter((s) => s.timestamp > lockTs).length;
   594→      // (The predictor filters internally, so future frames in the array
   595→      // are never used. This is structurally guaranteed.)
   596→      if (futureFrames > 0) {
   597→        // Not a violation — the predictor filters them out
   598→      }
   599→
   600→      // Validity check (FROZEN rules, determined BEFORE seeing result)
   601→      // BEFORE_SPIN is NOT a leakage violation — it's just INSUFFICIENT
   602→      let isValidLock: boolean;
   603→      let invalidReason = "";
   604→
   605→      if (isBeforeSpin) {
   606→        // Lock point is before movement started — INSUFFICIENT (not a violation)
   607→        isValidLock = false;
   608→        invalidReason = `BEFORE_SPIN — lock is ${((movementStart - lockTs) / 1000).toFixed(1)}s before movement start`;
   609→      } else {
   610→        // Lock is during the spin — check moving frames + confidence
   611→        isValidLock = movingFrames >= MIN_MOVING_FRAMES
   612→          && trackingConfidence >= MIN_TRACKING_CONFIDENCE
   613→          && isPreStop;
   614→
   615→        if (!isValidLock) {
   616→          if (movingFrames < MIN_MOVING_FRAMES) {
   617→            invalidReason = `INSUFFICIENT — only ${movingFrames} moving frames (need ${MIN_MOVING_FRAMES})`;
   618→          } else if (trackingConfidence < MIN_TRACKING_CONFIDENCE) {
   619→            invalidReason = `INSUFFICIENT — tracking confidence ${(trackingConfidence * 100).toFixed(0)}% < ${(MIN_TRACKING_CONFIDENCE * 100).toFixed(0)}%`;
   620→          }
   621→        }
   622→      }
   623→
   624→      // Generate prediction
   625→      const prediction = predictStoppingAngle(spinSnapshots, lockTs);
   626→
   627→      const actualSector = spin.actualSector;
   628→      const actualAngle = actualSector !== null
   629→        ? actualSector * SECTOR_WIDTH + SECTOR_WIDTH / 2
   630→        : null;
   631→
   632→      // Angular error (circular)
   633→      let angularError: number | null = null;
   634→      let sectorDistance: number | null = null;
   635→      if (prediction.isValid && actualAngle !== null) {
   636→        angularError = Math.abs(circularDiff(actualAngle, prediction.predictedStopAngle));
   637→        const d = Math.abs(prediction.predictedStopSector - (actualSector ?? 0));
   638→        sectorDistance = Math.min(d, 54 - d);
   639→      }
   640→
   641→      const hit = prediction.isValid && prediction.top4.includes(spin.actualOutcome);
   642→
   643→      // LEAKAGE CHECK 3: result must arrive after lock
   644→      if (spin.physicalSpinStop <= lockTs) {
   645→        resultLeakageViolations++;
   646→      }
   647→
   648→      if (!prediction.isValid) {
   649→        invalidPredictions++;
   650→      }
   651→
   652→      const result: SpinValidationResult = {
   653→        spinId: spin.spinId,
   654→        actualOutcome: spin.actualOutcome,
   655→        actualSector,
   656→        lockPoint: lp,
   657→        lockTimestamp: lockTs,
   658→        secondsSinceSpinStart: (lockTs - spin.physicalSpinStart) / 1000,
   659→        secondsBeforePhysicalStop: (spin.physicalSpinStop - lockTs) / 1000,
   660→        movingFrameCount: movingFrames,
   661→        trackingConfidence,
   662→        valid: isValidLock && prediction.isValid,
   663→        invalidReason: invalidReason || prediction.reason,
   664→        prediction,
   665→        hit,
   666→        predictedSector: prediction.predictedStopSector,
   667→        actualAngle,
   668→        predictedAngle: prediction.predictedStopAngle,
   669→        angularError,
   670→        sectorDistance,
   671→        predictedSectorProb: prediction.isValid
   672→          ? prediction.sectorProbabilities[prediction.predictedStopSector] ?? 0
   673→          : 0,
   674→        actualSectorProb: prediction.isValid && actualSector !== null
   675→          ? prediction.sectorProbabilities[actualSector] ?? 0
   676→          : 0,
   677→        leakage: { passed: true, violations: 0, details: "PASS (structurally guaranteed)" },
   678→      };
   679→
   680→      allResults.push(result);
   681→
   682→      // Arm comparison at primary LP
   683→      if (lp === primaryLP) {
   684→        if (prediction.isValid) {
   685→          videoTotal++;
   686→          if (hit) videoHits++;
   687→        } else {
   688→          videoInsufficient++;
   689→        }
   690→        theoTotal++;
   691→        if (["1", "2", "5", "10"].includes(spin.actualOutcome)) theoHits++;
   692→        histTotal++;
   693→        if (["1", "2", "5", "10"].includes(spin.actualOutcome)) histHits++;
   694→      }
   695→    }
   696→  }
   697→
   698→  // Build lock-point summaries
   699→  for (const lp of allLPs) {
   700→    const lpResults = allResults.filter((r) => r.lockPoint === lp);
   701→    const validResults = lpResults.filter((r) => r.valid && r.prediction.isValid);
   702→    const hits = validResults.filter((r) => r.hit).length;
   703→    const misses = validResults.length - hits;
   704→    const insufficient = lpResults.length - validResults.length;
   705→    const total = validResults.length;
   706→
   707→    const errors = validResults
   708→      .filter((r) => r.angularError !== null)
   709→      .map((r) => r.angularError!)
   710→      .sort((a, b) => a - b);
   711→
   712→    const sectorCorrect = validResults.filter(
   713→      (r) => r.actualSector !== null && r.predictedSector === r.actualSector,
   714→    ).length;
   715→
   716→    const top1Correct = validResults.filter((r) => {
   717→      if (!r.prediction.isValid) return false;
   718→      const sorted = Object.entries(r.prediction.outcomeProbabilities).sort((a, b) => b[1] - a[1]);
   719→      return sorted[0]?.[0] === r.actualOutcome;
   720→    }).length;
   721→
   722→    const bonusInclusion = validResults.filter((r) =>
   723→      r.prediction.top4.some((name) => BONUS_NAMES.includes(name)),
   724→    ).length;
   725→
   726→    const numberExclusion = validResults.filter((r) => {
   727→      const numbers = ["1", "2", "5", "10"];
   728→      return numbers.some((n) => !r.prediction.top4.includes(n));
   729→    }).length;
   730→
   731→    lockPointSummaries[lp] = {
   732→      lockPoint: lp,
   733→      sampleSize: total,
   734→      hits,
   735→      misses,
   736→      insufficientCount: insufficient,
   737→      hitRate: total > 0 ? hits / total : 0,
   738→      bonusInclusion,
   739→      numberExclusion,
   740→      avgConfidence: total > 0
   741→        ? validResults.reduce((s, r) => s + r.prediction.trackingConfidence, 0) / total
   742→        : 0,
   743→      avgTrackingQuality: total > 0
   744→        ? validResults.reduce((s, r) => s + r.prediction.signalQuality, 0) / total
   745→        : 0,
   746→      avgAngularError: errors.length > 0 ? errors.reduce((s, v) => s + v, 0) / errors.length : null,
   747→      medianAngularError: errors.length > 0 ? errors[Math.floor(errors.length / 2)] : null,
   748→      p90AngularError: errors.length > 0 ? errors[Math.floor(errors.length * 0.9)] : null,
   749→      sectorAccuracy: total > 0 ? sectorCorrect / total : 0,
   750→      top1OutcomeAccuracy: total > 0 ? top1Correct / total : 0,
   751→      avgUncertainty: total > 0
   752→        ? validResults.reduce((s, r) => s + r.prediction.angularUncertainty, 0) / total
   753→        : 0,
   754→    };
   755→  }
   756→
   757→  // Sensor quality breakdown (at primary LP)
   758→  const primaryResults = allResults.filter((r) => r.lockPoint === primaryLP && r.valid && r.prediction.isValid);
   759→  const high = primaryResults.filter((r) => r.prediction.trackingConfidence > 0.5);
   760→  const medium = primaryResults.filter((r) => r.prediction.trackingConfidence > 0.2 && r.prediction.trackingConfidence <= 0.5);
   761→  const low = primaryResults.filter((r) => r.prediction.trackingConfidence <= 0.2);
   762→
   763→  const sensorQuality = {
   764→    high: {
   765→      count: high.length,
   766→      hitRate: high.length > 0 ? high.filter((r) => r.hit).length / high.length : 0,
   767→    },
   768→    medium: {
   769→      count: medium.length,
   770→      hitRate: medium.length > 0 ? medium.filter((r) => r.hit).length / medium.length : 0,
   771→    },
   772→    low: {
   773→      count: low.length,
   774→      hitRate: low.length > 0 ? low.filter((r) => r.hit).length / low.length : 0,
   775→    },
   776→  };
   777→
   778→  const angularErrors = primaryResults
   779→    .filter((r) => r.angularError !== null)
   780→    .map((r) => r.angularError!);
   781→
   782→  const missDetails = allResults.filter((r) => !r.hit && r.valid && r.prediction.isValid);
   783→
   784→  // Spin duration stats
   785→  durations.sort((a, b) => a - b);
   786→  const movementDelaysSorted = [...movementDelays].sort((a, b) => a - b);
   787→
   788→  return {
   789→    totalSpins: spins.length,
   790→    matchedSpins: spins.filter((s) => s.actualOutcome !== null).length,
   791→    spinDurationStats: {
   792→      min: durations[0] ?? 0,
   793→      median: durations[Math.floor(durations.length / 2)] ?? 0,
   794→      mean: durations.length > 0 ? durations.reduce((s, v) => s + v, 0) / durations.length : 0,
   795→      max: durations[durations.length - 1] ?? 0,
   796→    },
   797→    movementStartStats: {
   798→      avgDelayFromSpinStart: movementDelaysSorted.length > 0
   799→        ? movementDelaysSorted.reduce((s, v) => s + v, 0) / movementDelaysSorted.length
   800→        : 0,
   801→    },
   802→    lockPointSummaries,
   803→    armComparison: {
   804→      video: {
   805→        hits: videoHits,
   806→        total: videoTotal,
   807→        hitRate: videoTotal > 0 ? videoHits / videoTotal : 0,
   808→        insufficient: videoInsufficient,
   809→      },
   810→      theoretical: {
   811→        hits: theoHits,
   812→        total: theoTotal,
   813→        hitRate: theoTotal > 0 ? theoHits / theoTotal : 0,
   814→      },
   815→      history: {
   816→        hits: histHits,
   817→        total: histTotal,
   818→        hitRate: histTotal > 0 ? histHits / histTotal : 0,
   819→      },
   820→    },
   821→    angularErrors,
   822→    missDetails,
   823→    leakageAudit: {
   824→      passed: futureFrameViolations === 0 && postStopViolations === 0 && resultLeakageViolations === 0,
   825→      futureFrameViolations,
   826→      postStopViolations,
   827→      resultLeakageViolations,
   828→      invalidPredictions,
   829→      details: futureFrameViolations === 0 && postStopViolations === 0 && resultLeakageViolations === 0
   830→        ? "PASS — no future-frame, post-stop, or result-leakage violations"
   831→        : `FAIL — ${futureFrameViolations} future-frame, ${postStopViolations} post-stop, ${resultLeakageViolations} result-leakage violations`,
   832→    },
   833→    sensorQuality,
   834→    modelVersion: "video-physics-v2.5",
   835→  };
   836→}

   838→// ============================================================
   839→// MISS CLASSIFICATION
   840→// ============================================================
   841→
   842→export type MissClassification =
   843→  | "SENSOR" // tracking confidence too low
   844→  | "TIMING" // lock point too early/late
   845→  | "SECTOR_MAP" // 54-sector mapping error
   846→  | "PHYSICS" // stopping angle prediction error
   847→  | "UNCERTAINTY" // angular uncertainty too high
   848→  | "MODEL"; // deceleration model limitation
   849→
   850→export interface MissRCA {
   851→  spinId: string;
   852→  lockPoint: string;
   853→  actual: string;
   854→  top4: string[];
   855→  actualSector: number | null;
   856→  predictedSector: number;
   857→  angularError: number | null;
   858→  velocity: number;
   859→  acceleration: number;
   860→  deceleration: number;
   861→  trackingConfidence: number;
   862→  uncertainty: number;
   863→  sectorProbabilityDistribution: Record<string, number>;
   864→  classification: MissClassification;
   865→  explanation: string;
   866→}
   867→
   868→export function classifyMiss(result: SpinValidationResult): MissRCA {
   869→  const p = result.prediction;
   870→  let classification: MissClassification;
   871→  let explanation: string;
   872→
   873→  if (p.trackingConfidence < 0.2) {
   874→    classification = "SENSOR";
   875→    explanation = `Tracking confidence ${(p.trackingConfidence * 100).toFixed(0)}% too low — sensor quality insufficient.`;
   876→  } else if (p.angularUncertainty > 90) {
   877→    classification = "UNCERTAINTY";
   878→    explanation = `Angular uncertainty ${p.angularUncertainty.toFixed(0)}° too high — prediction spread too wide.`;
   879→  } else if (result.angularError !== null && result.angularError > 60) {
   880→    classification = "PHYSICS";
   881→    explanation = `Angular error ${result.angularError.toFixed(0)}° — stopping angle prediction inaccurate. Deceleration model may not match actual wheel dynamics.`;
   882→  } else if (result.sectorDistance !== null && result.sectorDistance > 5) {
   883→    classification = "SECTOR_MAP";
   884→    explanation = `Sector distance ${result.sectorDistance} — predicted sector far from actual. 54-sector mapping may be incorrect.`;
   885→  } else if (p.deceleration < 10) {
   886→    classification = "TIMING";
   887→    explanation = `Deceleration ${p.deceleration.toFixed(1)}°/s² too low — lock point may be too early (wheel still accelerating).`;
   888→  } else {
   889→    classification = "MODEL";
   890→    explanation = `Deceleration model limitation — constant-deceleration assumption may not match actual wheel dynamics.`;
   891→  }
   892→
   893→  // Sector probability distribution (top 8)
   894→  const sectorDist: Record<string, number> = {};
   895→  if (p.isValid) {
   896→    const sorted = Object.entries(p.outcomeProbabilities).sort((a, b) => b[1] - a[1]);
   897→    for (const [name, prob] of sorted) {
   898→      sectorDist[name] = prob;
   899→    }
   900→  }
   901→
   902→  return {
   903→    spinId: result.spinId,
   904→    lockPoint: result.lockPoint,
   905→    actual: result.actualOutcome,
   906→    top4: p.top4,
   907→    actualSector: result.actualSector,
   908→    predictedSector: result.predictedSector,
   909→    angularError: result.angularError,
   910→    velocity: p.rawVelocity,
   911→    acceleration: p.acceleration,
   912→    deceleration: p.deceleration,
   913→    trackingConfidence: p.trackingConfidence,
   914→    uncertainty: p.angularUncertainty,
   915→    sectorProbabilityDistribution: sectorDist,
   916→    classification,
   917→    explanation,
   918→  };
   919→}
