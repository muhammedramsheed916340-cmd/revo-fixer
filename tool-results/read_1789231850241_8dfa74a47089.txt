     1→  1080→ * NEW: When `liveSpins` (REAL casino spins from CasinoScores API) is available
     2→  1081→ * with 20+ spins, the long-term frequency is computed from REAL casino data —
     3→  1082→ * NOT just the tiny user-verified round history. This produces data-driven,
     4→  1083→ * VARIED predictions instead of always picking the theoretical top-4 [1,2,5,10].
     5→  1084→ */
     6→  1085→function scoreCandidates(
     7→  1086→  rounds: RoundResult[],
     8→  1087→  dashboard: PerformanceDashboard,
     9→  1088→  prevPredNames: string[],
    10→  1089→  lastHit: boolean | null,
    11→  1090→  liveSpins: SpinData[] = [],
    12→  1091→  mode: EngineMode = "baseline",
    13→  1092→): CandidateScore[] {
    14→  1093→  // Get adaptive signal weights (learn from HIT/MISS history)
    15→  1094→  const sigWeights = dashboard.signalWeights ?? {};
    16→  1095→  const n = rounds.length;
    17→  1096→  const hist = rounds.map((r) => r.actualResult);
    18→  1097→
    19→  1098→  // Per-game frequency across ALL verified rounds (user history)
    20→  1099→  const freqAll = new Map<string, number>();
    21→  1100→  for (const g of GAMES) freqAll.set(g.name, 0);
    22→  1101→  for (const h of hist) freqAll.set(h.name, (freqAll.get(h.name) ?? 0) + 1);
    23→  1102→
    24→  1103→  // ===== REAL CASINO SPIN FREQUENCY (long-term prior from live data) =====
    25→  1104→  // Map live spins (aiStats sector names) to engine game names + count.
    26→  1105→  const liveN = liveSpins.length;
    27→  1106→  const liveFreqAll = new Map<string, number>();
    28→  1107→  for (const g of GAMES) liveFreqAll.set(g.name, 0);
    29→  1108→  for (const s of liveSpins) {
    30→  1109→    const gameName = SPIN_TO_GAME_NAME[s.sector] ?? s.sector;
    31→  1110→    if (liveFreqAll.has(gameName)) {
    32→  1111→      liveFreqAll.set(gameName, (liveFreqAll.get(gameName) ?? 0) + 1);
    33→  1112→    }
    34→  1113→  }
    35→  1114→  // Use real casino frequency as long-term prior when we have 20+ real spins.
    36→  1115→  // Otherwise fall back to theoretical probability.
    37→  1116→  const useLivePrior = liveN >= 20;
    38→  1117→  const livePriorFor = (gameName: string): number => {
    39→  1118→    if (!useLivePrior) return THEORETICAL[gameName] ?? 0.1;
    40→  1119→    return liveN > 0 ? (liveFreqAll.get(gameName) ?? 0) / liveN : (THEORETICAL[gameName] ?? 0.1);
    41→  1120→  };
    42→  1121→
    43→  1122→  // ===== RECENT LIVE SPIN FREQUENCY (last 10 real spins) =====
    44→  1123→  const liveRecent = liveSpins.slice(0, 10); // newest first in API response
    45→  1124→  const liveRecentN = liveRecent.length;
    46→  1125→  const liveRecentFreq = new Map<string, number>();
    47→  1126→  for (const g of GAMES) liveRecentFreq.set(g.name, 0);
    48→  1127→  for (const s of liveRecent) {
    49→  1128→    const gameName = SPIN_TO_GAME_NAME[s.sector] ?? s.sector;
    50→  1129→    if (liveRecentFreq.has(gameName)) {
    51→  1130→      liveRecentFreq.set(gameName, (liveRecentFreq.get(gameName) ?? 0) + 1);
    52→  1131→    }
    53→  1132→  }
    54→  1133→
    55→  1134→  // Per-game frequency in recent 10 USER-VERIFIED rounds (short-term)
    56→  1135→  const recentHist = hist.slice(-10);
    57→  1136→  const recentFreq = new Map<string, number>();
    58→  1137→  for (const g of GAMES) recentFreq.set(g.name, 0);
    59→  1138→  for (const h of recentHist) recentFreq.set(h.name, (recentFreq.get(h.name) ?? 0) + 1);
    60→  1139→
    61→  1140→  // Gap analysis per game
    62→  1141→  const gaps: Record<string, number> = {};
    63→  1142→  const gapHistories: Record<string, number[]> = {};
    64→  1143→  for (const g of GAMES) { gaps[g.name] = n; gapHistories[g.name] = []; }
    65→  1144→  let lastHitIdx: Record<string, number> = {};
    66→  1145→  for (const g of GAMES) lastHitIdx[g.name] = -1;
    67→  1146→  for (let i = 0; i < n; i++) {
    68→  1147→    const name = hist[i].name;
    69→  1148→    if (lastHitIdx[name] >= 0) {
    70→  1149→      gapHistories[name].push(i - lastHitIdx[name]);
    71→  1150→    }
    72→  1151→    lastHitIdx[name] = i;
    73→  1152→  }
    74→  1153→  for (const g of GAMES) {
    75→  1154→    gaps[g.name] = lastHitIdx[g.name] >= 0 ? n - 1 - lastHitIdx[g.name] : n;
    76→  1155→  }
    77→  1156→
    78→  1157→  // Adaptive weighting between long-term and recent
    79→  1158→  const adaptW = dashboard.adaptiveWeight;
    80→  1159→  const longW = 1 - adaptW;
    81→  1160→
    82→  1161→  // Per-game stability (variance of recent gaps → inverse)
    83→  1162→  const recentUnique = new Set(recentHist.map((g) => g.name)).size;
    84→  1163→  const stability = recentUnique >= 4 ? "STABLE" : recentUnique >= 2 ? "VOLATILE" : "STREAK";
    85→  1164→
    86→  1165→  // ===== REPEAT-PATTERN ANALYSIS (data-based, NOT blind carryover) =====
    87→  1166→  // Analyze the historical tendency of each outcome to REPEAT in the next
    88→  1167→  // round. This is NOT "last hit → predict same again". Instead, we measure:
    89→  1168→  //   P(next = X | current = X)  from ALL historical consecutive pairs.
    90→  1169→  // If the last actual result was X, and X has a historically HIGH repeat
    91→  1170→  // rate, then X gets a MILD evidence boost (capped). If X has a LOW repeat
    92→  1171→  // rate, no penalty — we just don't boost. This is pure statistical
    93→  1172→  // evidence, not a "last hit shortcut".
    94→  1173→  //
    95→  1174→  // We use BOTH the user-verified round history AND the live casino spins
    96→  1175→  // (mapped to game names) for a larger, more reliable sample.
    97→  1176→  const lastActualName = n > 0 ? hist[n - 1].name : null;
    98→  1177→  const lastLiveSpinName = liveN > 0
    99→  1178→    ? (SPIN_TO_GAME_NAME[liveSpins[0].sector] ?? null)
   100→  1179→    : null;
   101→  1180→  // Prefer the most recent data point (live spins are newer than user rounds).
   102→  1181→  const lastResultName = lastLiveSpinName ?? lastActualName;
   103→  1182→
   104→  1183→  // Build a combined consecutive-pair sequence from both sources.
   105→  1184→  // Live spins are newest-first; reverse to oldest-first for pair analysis.
   106→  1185→  const liveGameSeq = liveSpins
   107→  1186→    .map((s) => SPIN_TO_GAME_NAME[s.sector] ?? null)
   108→  1187→    .filter((x): x is string => x !== null)
   109→  1188→    .reverse(); // oldest-first
   110→  1189→  const userGameSeq = hist.map((h) => h.name);
   111→  1190→  const combinedSeq = [...userGameSeq, ...liveGameSeq];
   112→  1191→
   113→  1192→  // Count repeat pairs: P(next = X | current = X)
   114→  1193→  const repeatStats: Record<string, { currentCount: number; repeatCount: number; rate: number }> = {};
   115→  1194→  for (const g of GAMES) repeatStats[g.name] = { currentCount: 0, repeatCount: 0, rate: 0 };
   116→  1195→  for (let i = 0; i < combinedSeq.length - 1; i++) {
   117→  1196→    const cur = combinedSeq[i];
   118→  1197→    const next = combinedSeq[i + 1];
   119→  1198→    if (repeatStats[cur]) {
   120→  1199→      repeatStats[cur].currentCount++;
   121→  1200→      if (next === cur) repeatStats[cur].repeatCount++;
   122→  1201→    }
   123→  1202→  }
   124→  1203→  for (const g of GAMES) {
   125→  1204→    const rs = repeatStats[g.name];
   126→  1205→    rs.rate = rs.currentCount >= 3 ? rs.repeatCount / rs.currentCount : 0;
   127→  1206→  }
   128→  1207→  // Overall average repeat rate (for comparison baseline).
   129→  1208→  const totalRepeatCur = Object.values(repeatStats).reduce((s, r) => s + r.currentCount, 0);
   130→  1209→  const totalRepeatCount = Object.values(repeatStats).reduce((s, r) => s + r.repeatCount, 0);
   131→  1210→  const baselineRepeatRate = totalRepeatCur >= 5 ? totalRepeatCount / totalRepeatCur : 0;
   132→  1211→
   133→  1212→  // ===== BONUS CLUSTER DETECTION (bonus-aware scoring) =====
   134→  1213→  // Analyze whether bonus outcomes are: increasing, decreasing, clustering,
   135→  1214→  // appearing after specific sequences, repeating, or showing unusual recent
   136→  1215→  // frequency. This is NOT "bonus appeared → predict bonus next". Instead,
   137→  1216→  // we measure the CURRENT bonus activity level vs the long-term baseline.
   138→  1217→  // If bonus activity is elevated, bonus outcomes get a mild evidence boost
   139→  1218→  // (capped). This ensures the engine does NOT have a bonus blind spot.
   140→  1219→  const isBonusGame = (name: string): boolean => BONUS_NAMES.includes(name);
   141→  1220→
   142→  1221→  // Bonus frequency in combined sequence (long-term)
   143→  1222→  const combinedBonusCount = combinedSeq.filter(isBonusGame).length;
   144→  1223→  const combinedBonusRate = combinedSeq.length > 0 ? combinedBonusCount / combinedSeq.length : 0;
   145→  1224→  // Bonus frequency in recent 10 of combined sequence (short-term)
   146→  1225→  const recentCombined = combinedSeq.slice(-10);
   147→  1226→  const recentBonusCount = recentCombined.filter(isBonusGame).length;
   148→  1227→  const recentBonusRate = recentCombined.length > 0 ? recentBonusCount / recentCombined.length : 0;
   149→  1228→  // Bonus activity trend: recent vs long-term
   150→  1229→  const bonusTrend = recentBonusRate - combinedBonusRate; // positive = increasing
   151→  1230→  // Bonus clustering: count bonus "bursts" (2+ bonuses within 3 spins)
   152→  1231→  let bonusBursts = 0;
   153→  1232→  for (let i = 0; i < combinedSeq.length - 2; i++) {
   154→  1233→    const window = combinedSeq.slice(i, i + 3);
   155→  1234→    const bonusInWindow = window.filter(isBonusGame).length;
   156→  1235→    if (bonusInWindow >= 2) bonusBursts++;
   157→  1236→  }
   158→  1237→  // Is a bonus currently "active" (appeared in last 3 spins)?
   159→  1238→  const last3 = combinedSeq.slice(-3);
   160→  1239→  const bonusInLast3 = last3.filter(isBonusGame).length;
   161→  1240→  const bonusActive = bonusInLast3 > 0;
   162→  1241→  // Per-bonus-game recent frequency (last 10 combined)
   163→  1242→  const bonusRecentFreq: Record<string, number> = {};
   164→  1243→  for (const b of BONUS_NAMES) bonusRecentFreq[b] = 0;
   165→  1244→  for (const name of recentCombined) {
   166→  1245→    if (isBonusGame(name)) bonusRecentFreq[name] = (bonusRecentFreq[name] ?? 0) + 1;
   167→  1246→  }
   168→  1247→
   169→  1248→  const scores: CandidateScore[] = [];
   170→  1249→
   171→  1250→  for (let i = 0; i < GAMES.length; i++) {
   172→  1251→    const g = GAMES[i];
   173→  1252→    const theo = THEORETICAL[g.name] ?? 0.1;
   174→  1253→    // LONG-TERM PRIOR: prefer REAL casino spin frequency when available (20+
   175→  1254→    // real spins), else theoretical probability, else tiny user history.
   176→  1255→    const livePrior = livePriorFor(g.name);
   177→  1256→    const userLongFreq = n > 0 ? (freqAll.get(g.name) ?? 0) / n : 0;
   178→  1257→    // Blend user-verified long-term with live-prior when both exist.
   179→  1258→    // Live prior dominates (real casino data) but user history adds weight.
   180→  1259→    const longFreq = useLivePrior
   181→  1260→      ? livePrior * 0.7 + userLongFreq * 0.3
   182→  1261→      : userLongFreq;
   183→  1262→    // RECENT frequency: prefer live recent (last 10 real spins) when available;
   184→  1263→    // it's far more responsive than the tiny user recent slice.
   185→  1264→    const liveRecFreq = liveRecentN > 0 ? (liveRecentFreq.get(g.name) ?? 0) / liveRecentN : 0;
   186→  1265→    const userRecFreq = recentHist.length > 0 ? (recentFreq.get(g.name) ?? 0) / recentHist.length : 0;
   187→  1266→    const recFreq = useLivePrior
   188→  1267→      ? liveRecFreq * 0.6 + userRecFreq * 0.4
   189→  1268→      : userRecFreq;
   190→  1269→    const trend = recFreq - longFreq;
   191→  1270→    const gap = gaps[g.name];
   192→  1271→    const gapHistory = gapHistories[g.name];
   193→  1272→    const avgGap = gapHistory.length > 0
   194→  1273→      ? gapHistory.reduce((s, x) => s + x, 0) / gapHistory.length
   195→  1274→      : n / Math.max(freqAll.get(g.name) ?? 1, 1);
   196→  1275→    const isOverdue = avgGap > 0 && gap > avgGap * 1.5;
   197→  1276→    const isHot = longFreq > theo * 1.3 && (n >= 5 || liveN >= 20);
   198→  1277→    const isCold = longFreq < theo * 0.5 && (n >= 5 || liveN >= 20);
   199→  1278→
   200→  1279→    // Signal-wise historical performance (when this game was predicted)
   201→  1280→    const sw = dashboard.signalWiseHitRate[g.name];
   202→  1281→    const sampleSize = sw.predicted;
   203→  1282→    const signalHitRate = sw.rate;
   204→  1283→    const wilsonLower = sw.wilsonLower; // sample-size-aware confidence
   205→  1284→
   206→  1285→    // Volatility: variance of gap history
   207→  1286→    let volatility = 0;
   208→  1287→    if (gapHistory.length >= 2) {
   209→  1288→      const mean = gapHistory.reduce((s, x) => s + x, 0) / gapHistory.length;
   210→  1289→      volatility = gapHistory.reduce((s, x) => s + (x - mean) ** 2, 0) / gapHistory.length;
   211→  1290→    }
   212→  1291→
   213→  1292→    const signals: string[] = [];
   214→  1293→
   215→  1294→    // ===== SAMPLE-SIZE STABILIZATION (Bayesian / Laplace shrinkage) =====
   216→  1295→    // CRITICAL: Small-sample observations must NOT create extreme deviations.
   217→  1296→    //
   218→  1297→    // k=30 (increased from 20): rare outcomes need MORE evidence before their
   219→  1298→    // deviation can significantly affect the score. At k=30:
   220→  1299→    //   N=30, CRAZY TIME count=1, prior=1.85%:
   221→  1300→    //     smoothed = (1 + 30*0.0185) / 60 = 1.09% → dev=-41% (was +48% at k=20)
   222→  1301→    //   N=30, "1" count=12, prior=38.89%:
   223→  1302→    //     smoothed = (12 + 30*0.389) / 60 = 39.45% → dev=+1.4% (barely changed)
   224→  1303→    //
   225→  1304→    // This ensures rare outcomes need substantially more observations before
   226→  1305→    // they can displace high-prior common outcomes like "1" (38.89%) or "2" (24.07%).
   227→  1306→    const SHRINKAGE_K = 30;
   228→  1307→
   229→  1308→    // Compute observed count and total N for this outcome.
   230→  1309→    // Use the larger of (user rounds, live spins) as the sample.
   231→  1310→    const userCount = freqAll.get(g.name) ?? 0;
   232→  1311→    const liveCount = liveFreqAll.get(g.name) ?? 0;
   233→  1312→    // Total sample N = combined user rounds + live spins
   234→  1313→    const sampleN = n + liveN;
   235→  1314→    // Combined observed count (user + live, weighted by their respective N)
   236→  1315→    const combinedCount = userCount + liveCount;
   237→  1316→
   238→  1317→    // Laplace-smoothed frequency
   239→  1318→    const smoothedFreq = sampleN > 0
   240→  1319→      ? (combinedCount + SHRINKAGE_K * theo) / (sampleN + SHRINKAGE_K)
   241→  1320→      : theo;
   242→  1321→
   243→  1322→    // Raw (unstabilized) deviation — for debug display only
   244→  1323→    const rawBlendedFreq = (longFreq * longW + recFreq * adaptW);
   245→  1324→    const rawDeviation = theo > 0 ? (rawBlendedFreq - theo) / theo : 0;
   246→  1325→
   247→  1326→    // Stabilized deviation — uses the Laplace-smoothed frequency
   248→  1327→    const stabilizedDeviation = theo > 0 ? (smoothedFreq - theo) / theo : 0;
   249→  1328→
   250→  1329→    // Cap stabilized deviation to prevent extreme swings: [-0.6, +2.0]
   251→  1330→    const cappedDeviation = Math.max(-0.6, Math.min(2.0, stabilizedDeviation));
   252→  1331→
   253→  1332→    // ===== EXPERIMENTAL: RARE-OUTCOME EVIDENCE RELIABILITY LAYER =====
   254→  1333→    // (ACTIVE only in mode === "experimental"; baseline passes through unchanged)
   255→  1334→    //
   256→  1335→    // A small sample of rare-outcome appearances (e.g. PACHINKO 3× in 50
   257→  1336→    // rounds) can generate a large RELATIVE deviation (+39%) that, via
   258→  1337→    // evidenceScore = 1 + dev, displaces a substantially higher-prior
   259→  1338→    // number outcome. This is statistically unsound: the relative
   260→  1339→    // deviation of a rare outcome has much higher variance per observation
   261→  1340→    // than a common outcome, so the same count implies far less confidence.
   262→  1341→    //
   263→  1342→    // Reliability factor (continuous, generic, NO hard cutoff):
   264→  1343→    //   reliability = effectiveSampleSize / (effectiveSampleSize + RELIABILITY_K)
   265→  1344→    //
   266→  1345→    //   effectiveSampleSize = combinedCount (user rounds + live spins that
   267→  1346→    //   landed on this outcome — the independent observations supporting
   268→  1347→    //   THIS outcome's deviation estimate).
   269→  1348→    //
   270→  1349→    // Applied ONLY to the POSITIVE portion of the deviation:
   271→  1350→    //   - Positive dev (outcome appeared MORE than prior) → dampened by
   272→  1351→    //     reliability. A rare outcome needs many independent observations
   273→  1352→    //     before its positive evidence can dominate a high-prior outcome.
   274→  1353→    //   - Negative dev (outcome appeared LESS than prior) → passes through
   275→  1354→    //     UNCHANGED. We must NOT inflate a rare outcome that hasn't
   276→  1355→    //     appeared (that would push never-seen rare outcomes INTO Top-4,
   277→  1356→    //     the opposite of the goal). The prior term already keeps them low.
   278→  1357→    //
   279→  1358→    // This is NOT recency chasing: a single recent appearance does not
   280→  1359→    // move reliability much (count 0→1: r 0→0.09). Genuine repeated
   281→  1360→    // evidence accumulates count → reliability rises → evidence reaches
   282→  1361→    // full strength. A rare outcome with 8+ appearances CAN still enter.
   283→  1362→    const experimentalActive = mode === "experimental";
   284→  1363→    const effectiveSampleSize = combinedCount; // independent observations
   285→  1364→    const reliability = experimentalActive
   286→  1365→      ? effectiveSampleSize / (effectiveSampleSize + RELIABILITY_K)
   287→  1366→      : 1.0; // baseline: full strength (no reliability discount)
   288→  1367→    // Dampen positive deviation only; keep negative deviation as-is.
   289→  1368→    const reliableDeviation = experimentalActive
   290→  1369→      ? (cappedDeviation > 0
   291→  1370→          ? cappedDeviation * reliability
   292→  1371→          : cappedDeviation)
   293→  1372→      : cappedDeviation; // baseline: identity
   294→  1373→
   295→  1374→    // ===== BASE SCORE — BALANCED EVIDENCE + PRIOR =====
   296→  1375→    // CRITICAL FIX: The previous 85% evidence / 15% prior ratio allowed rare
   297→  1376→    // outcomes (CRAZY TIME 1.85%) to displace common outcomes ("1" 38.89%)
   298→  1377→    // after just 1-2 appearances. This caused the model to UNDERPERFORM the
   299→  1378→    // simple theoretical [1,2,5,10] baseline (53% vs 88%).
   300→  1379→    //
   301→  1380→    // NEW RATIO: 50% evidence / 50% prior.
   302→  1381→    //
   303→  1382→    // This means:
   304→  1383→    //   - Common outcomes with NEUTRAL evidence keep ~50% of their prior weight
   305→  1384→    //   - Rare outcomes need MUCH stronger relative evidence to overcome
   306→  1385→    //     the prior advantage of common outcomes
   307→  1386→    //   - A rare outcome appearing 1-2× in 30 spins will NOT displace "1" or "2"
   308→  1387→    //   - A rare outcome appearing 5+× in 30 spins CAN still enter Top-4
   309→  1388→    //
   310→  1389→    // Example with N=30, k=30:
   311→  1390→    //   "1" (prior 38.89%): smoothed=39.45%, dev=+1.4% → evidence=1.014
   312→  1391→    //     score = 1.014*0.50 + 0.389*0.50 = 0.702
   313→  1392→    //   CRAZY TIME (prior 1.85%): smoothed=1.09%, dev=-41% → evidence=0.59
   314→  1393→    //     score = 0.59*0.50 + 0.0185*0.50 = 0.304
   315→  1394→    //   → "1" (0.702) >> CRAZY TIME (0.304) — correct!
   316→  1395→    //
   317→  1396→    //   CRAZY TIME appearing 3× in 30: smoothed=(3+30*0.0185)/60=5.93%, dev=+221%
   318→  1397→    //     evidence=3.21, score=3.21*0.50+0.0185*0.50=1.614
   319→  1398→    //   "1" appearing 12× in 30: score=0.702
   320→  1399→    //   → CRAZY TIME (1.614) > "1" (0.702) — rare outcome CAN enter when justified ✓
   321→  1400→    //
   322→  1401→    // EXPERIMENTAL mode: evidenceScore uses reliableDeviation (positive dev
   323→  1402→    // dampened by sample-size reliability). Baseline mode: reliableDeviation
   324→  1403→    // === cappedDeviation (identity), so behavior is BIT-FOR-BIT identical
   325→  1404→    // to the frozen k=30 baseline. This guarantees a clean A/B comparison.
   326→  1405→    const evidenceScore = 1 + reliableDeviation;
   327→  1406→    let score = evidenceScore * 0.50 + theo * 0.50;
   328→  1407→    if (sampleN === 0) score = theo; // no data at all → pure theoretical
   329→  1408→
   330→  1409→    // ===== FACTOR 1: Recent active (mild adaptive signal — NOT a chase) =====
   331→  1410→    // Appearing more than 80% of theoretical recently → mild evidence the wheel
   332→  1411→    // is currently favouring it (regime). Capped to prevent hot-number chasing.
   333→  1412→    // Adaptive weight applied from MISS-feedback calibration.
   334→  1413→    if (recFreq > livePrior * 0.8 && (n >= 5 || liveN >= 20)) {
   335→  1414→      const w = sigWeights["recent-active"] ?? 1.0;
   336→  1415→      score *= 1 + (0.10 - 1) * (1 - w) + 0.10 * w; // = 1.10 * w + (1 - w)
   337→  1416→      signals.push("recent-active");
   338→  1417→    }
   339→  1418→
   340→  1419→    // ===== FACTOR 2: Trend alignment (adaptive weighting) =====
   341→  1420→    // Recent vs long-term delta — mild evidence of regime shift. NOT a chase.
   342→  1421→    // Adaptive weight applied from MISS-feedback calibration.
   343→  1422→    if (trend > 0.05 && (n >= 10 || liveN >= 20)) {
   344→  1423→      const w = sigWeights["trending-up"] ?? 1.0;
