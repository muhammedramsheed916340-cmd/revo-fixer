   660→        // 🔐 LOGIN
   661→        // ============================================================
   662→        function login() {
   663→            const licenseKey = document.getElementById('licenseKey').value.trim().toUpperCase();
   664→            const deviceId = getDeviceId();
   665→
   666→            if (!licenseKey) {
   667→                showMessage('Please enter license key!', 'error');
   668→                return;
   669→            }
   670→
   671→            showMessage('Verifying...', 'success');
   672→
   673→            db.ref(`securityCodes/${licenseKey}`).once('value').then(snapshot => {
   674→                if (!snapshot.exists()) {
   675→                    showMessage('Invalid license key!', 'error');
   676→                    return;
   677→                }
   678→
   679→                const keyData = snapshot.val();
   680→
   681→                if (keyData.status === 'banned') {
   682→                    showMessage('License key blocked. Contact support.', 'error');
   683→                    return;
   684→                }
   685→
   686→                if (keyData.status === 'reset_required') {
   687→                    showMessage('Your key requires reset. Contact @RevoAgent to reset your key', 'error');
   688→                    return;
   689→                }
   690→
   691→                let deviceLogins = keyData.deviceLogins || {};
   692→                let currentDeviceAttempts = deviceLogins[deviceId] || 0;
   693→                let userId = keyData.usedBy;
   694→                const now = Date.now();
   695→
   696→                if (currentDeviceAttempts >= 3) {
   697→                    db.ref(`securityCodes/${licenseKey}`).update({
   698→                        status: 'reset_required',
   699→                        resetRequestedAt: now,
   700→                        resetRequestedBy: deviceId
   701→                    });
   702→                    showMessage('Your key is used on another device. Contact @RevoAgent to reset your key', 'error');
   703→                    return;
   704→                }
   705→
   706→                if (!userId) {
   707→                    userId = 'user_' + now + '_' + Math.random().toString(36).substring(2, 8);
   708→                    
   709→                    const packageInfo = keyData.package || { 
   710→                        name: '1 Hour Package', 
   711→                        price: 2000, 
   712→                        hours: 1
   713→                    };
   714→                    
   715→                    const endTime = now + (packageInfo.hours * 60 * 60 * 1000);
   716→
   717→                    const userData = {
   718→                        username: 'User',
   719→                        balance: 0,
   720→                        createdAt: now,
   721→                        package: {
   722→                            name: packageInfo.name,
   723→                            price: packageInfo.price,
   724→                            hours: packageInfo.hours,
   725→                            status: 'approved',
   726→                            startTime: now,
   727→                            endTime: endTime
   728→                        },
   729→                        usedKey: licenseKey,
   730→                        deviceId: deviceId,
   731→                        lastLogin: now,
   732→                        activity: [{
   733→                            text: '✅ Account created with license key',
   734→                            time: now
   735→                        }]
   736→                    };
   737→                    
   738→                    db.ref(`users/${userId}`).set(userData);
   739→
   740→                    deviceLogins[deviceId] = 1;
   741→                    db.ref(`securityCodes/${licenseKey}`).update({
   742→                        status: 'used',
   743→                        usedBy: userId,
   744→                        usedAt: now,
   745→                        deviceLogins: deviceLogins,
   746→                        totalDevices: Object.keys(deviceLogins).length
   747→                    });
   748→                    
   749→                    showMessage('✅ License activated! Welcome!', 'success');
   750→                    
   751→                } else {
   752→                    if (deviceLogins[deviceId] === undefined) {
   753→                        deviceLogins[deviceId] = 1;
   754→                    } else {
   755→                        deviceLogins[deviceId] = currentDeviceAttempts + 1;
   756→                    }
   757→                    
   758→                    db.ref(`securityCodes/${licenseKey}`).update({
   759→                        lastLogin: now,
   760→                        deviceLogins: deviceLogins,
   761→                        lastDeviceId: deviceId
   762→                    });
   763→
   764→                    db.ref(`users/${userId}`).update({
   765→                        lastLogin: now,
   766→                        deviceId: deviceId
   767→                    });
   768→                    
   769→                    db.ref(`users/${userId}/activity`).push({
   770→                        text: `🔑 Login attempt #${deviceLogins[deviceId]} from this device`,
   771→                        time: now
   772→                    });
   773→                    
   774→                    showMessage('✅ Login successful!', 'success');
   775→                }
   776→
   777→                localStorage.setItem('userId', userId);
   778→                localStorage.setItem('licenseKey', licenseKey);
   779→
   780→                setTimeout(() => {
   781→                    window.location.href = 'dashboard.html';
   782→                }, 1500);
   783→
   784→            }).catch(error => {
   785→                console.error('Login error:', error);
   786→                showMessage('Error: ' + error.message, 'error');
   787→            });
   788→        }
