     1→   982→ * After a HIT: if a signal was active on the CORRECT prediction, increase by 1%.
     2→   983→ * Capped to prevent any single signal from dominating or disappearing.
     3→   984→ */
     4→   985→function getAdaptiveSignalWeights(rounds: RoundResult[]): Record<string, number> {
     5→   986→  const weights: Record<string, number> = {};
     6→   987→  const SIGNAL_NAMES = [
     7→   988→    "recent-active", "trending-up", "trending-down", "pattern-stable",
     8→   989→    "verified", "high-volatility", "repeat-supported", "repeat-unlikely",
     9→   990→    "bonus-phase-risk", "persistence-penalty", "anomaly-weighted", "shift-adaptive",
    10→   991→  ];
    11→   992→  for (const s of SIGNAL_NAMES) weights[s] = 1.0;
    12→   993→
    13→   994→  if (rounds.length < 5) return weights; // need minimum data
    14→   995→
    15→   996→  // Track signal presence per round (from stored round metadata — we approximate
    16→   997→  // by checking if the actual result was predicted and had signals)
    17→   998→  // Since we don't store per-signal history, we use a simplified heuristic:
    18→   999→  // Count how many MISSes vs HITs occurred, and adjust weights gradually.
    19→  1000→  const recent20 = rounds.slice(-20);
    20→  1001→  const recentHits = recent20.filter((r) => r.hit).length;
    21→  1002→  const recentMisses = recent20.length - recentHits;
    22→  1003→  const hitRate = recentHits / recent20.length;
    23→  1004→
    24→  1005→  // If hit rate is low, slightly reduce reliance on trend/recent signals
    25→  1006→  // (they may be chasing noise). If hit rate is high, keep them strong.
    26→  1007→  if (recentMisses > recentHits * 1.5) {
    27→  1008→    // More misses than hits → reduce trend/recent signal weights slightly
    28→  1009→    weights["trending-up"] = Math.max(0.7, 1.0 - recentMisses * 0.01);
    29→  1010→    weights["trending-down"] = Math.max(0.7, 1.0 - recentMisses * 0.01);
    30→  1011→    weights["recent-active"] = Math.max(0.7, 1.0 - recentMisses * 0.005);
    31→  1012→  } else if (recentHits > recentMisses * 1.5) {
    32→  1013→    // More hits → slightly boost
    33→  1014→    weights["trending-up"] = Math.min(1.2, 1.0 + recentHits * 0.005);
    34→  1015→    weights["recent-active"] = Math.min(1.2, 1.0 + recentHits * 0.005);
    35→  1016→  }
    36→  1017→
    37→  1018→  return weights;
    38→  1019→}
    39→  1020→
    40→  1021→/**
    41→  1022→ * Analyze the last MISS: what was the actual result, what rank was it,
    42→  1023→ * and which signals may have suppressed it.
    43→  1024→ */
    44→  1025→function getLastMissAnalysis(rounds: RoundResult[]): {
    45→  1026→  actualResult: string;
    46→  1027→  actualRank: number;
    47→  1028→  actualScore: number;
    48→  1029→  predictedNames: string[];
    49→  1030→  signalsSuppressedActual: string[];
    50→  1031→  signalsBoostedFailed: string[];
    51→  1032→} | null {
    52→  1033→  if (rounds.length === 0) return null;
    53→  1034→  const last = rounds[rounds.length - 1];
    54→  1035→  if (last.hit) return null; // last round was HIT, no MISS analysis
    55→  1036→
    56→  1037→  const actualName = last.actualResult.name;
    57→  1038→  const predictedNames = last.prediction.map((p) => p.game.name);
    58→  1039→
    59→  1040→  // The actual result was NOT in the prediction set (it was a MISS).
    60→  1041→  // Analyze why: which signals may have suppressed it?
    61→  1042→  const suppressed: string[] = [];
    62→  1043→  const boosted: string[] = [];
    63→  1044→
    64→  1045→  // Check if the actual result had low recent frequency (suppressed by recency)
    65→  1046→  const actualRecentCount = rounds.slice(-10).filter((r) => r.actualResult.name === actualName).length;
    66→  1047→  if (actualRecentCount <= 1) suppressed.push("low-recent-frequency");
    67→  1048→
    68→  1049→  // Check if the actual result had high volatility (suppressed by volatility penalty)
    69→  1050→  if (last.recalibrated) suppressed.push("recalibration-applied");
    70→  1051→
    71→  1052→  // Check if the actual result was a bonus (suppressed by base prior)
    72→  1053→  if (BONUS_NAMES.includes(actualName)) suppressed.push("low-base-prior");
    73→  1054→
    74→  1055→  // Check if predicted outcomes had high recent frequency (boosted by recency)
    75→  1056→  for (const name of predictedNames) {
    76→  1057→    const count = rounds.slice(-10).filter((r) => r.actualResult.name === name).length;
    77→  1058→    if (count >= 3) boosted.push(`${name}-recent-active`);
    78→  1059→  }
    79→  1060→
    80→  1061→  // Approximate rank: actual result was not in top-4, so rank > 4
    81→  1062→  const actualRank = predictedNames.includes(actualName) ? 1 : 5; // simplified
    82→  1063→
    83→  1064→  return {
    84→  1065→    actualResult: actualName,
    85→  1066→    actualRank,
    86→  1067→    actualScore: 0, // not stored per-round; would need re-computation
    87→  1068→    predictedNames,
    88→  1069→    signalsSuppressedActual: suppressed,
    89→  1070→    signalsBoostedFailed: boosted,
    90→  1071→  };
    91→  1072→}
    92→  1073→
    93→  1074→// ============================================================
    94→  1075→// CANDIDATE SCORING — STRONGEST EVIDENCE-BASED SIGNAL
    95→  1076→// ============================================================
    96→  1077→/**
    97→  1078→ * Score all 8 game candidates using a multi-signal evidence engine.
    98→  1079→ *
    99→  1080→ * NEW: When `liveSpins` (REAL casino spins from CasinoScores API) is available
   100→  1081→ * with 20+ spins, the long-term frequency is computed from REAL casino data —
   101→  1082→ * NOT just the tiny user-verified round history. This produces data-driven,
   102→  1083→ * VARIED predictions instead of always picking the theoretical top-4 [1,2,5,10].
   103→  1084→ */
   104→  1085→function scoreCandidates(
   105→  1086→  rounds: RoundResult[],
   106→  1087→  dashboard: PerformanceDashboard,
   107→  1088→  prevPredNames: string[],
   108→  1089→  lastHit: boolean | null,
   109→  1090→  liveSpins: SpinData[] = [],
   110→  1091→  mode: EngineMode = "baseline",
   111→  1092→): CandidateScore[] {
   112→  1093→  // Get adaptive signal weights (learn from HIT/MISS history)
   113→  1094→  const sigWeights = dashboard.signalWeights ?? {};
   114→  1095→  const n = rounds.length;
   115→  1096→  const hist = rounds.map((r) => r.actualResult);
   116→  1097→
   117→  1098→  // Per-game frequency across ALL verified rounds (user history)
   118→  1099→  const freqAll = new Map<string, number>();
   119→  1100→  for (const g of GAMES) freqAll.set(g.name, 0);
   120→  1101→  for (const h of hist) freqAll.set(h.name, (freqAll.get(h.name) ?? 0) + 1);
   121→  1102→
   122→  1103→  // ===== REAL CASINO SPIN FREQUENCY (long-term prior from live data) =====
   123→  1104→  // Map live spins (aiStats sector names) to engine game names + count.
   124→  1105→  const liveN = liveSpins.length;
   125→  1106→  const liveFreqAll = new Map<string, number>();
   126→  1107→  for (const g of GAMES) liveFreqAll.set(g.name, 0);
   127→  1108→  for (const s of liveSpins) {
   128→  1109→    const gameName = SPIN_TO_GAME_NAME[s.sector] ?? s.sector;
   129→  1110→    if (liveFreqAll.has(gameName)) {
   130→  1111→      liveFreqAll.set(gameName, (liveFreqAll.get(gameName) ?? 0) + 1);
   131→  1112→    }
   132→  1113→  }
   133→  1114→  // Use real casino frequency as long-term prior when we have 20+ real spins.
   134→  1115→  // Otherwise fall back to theoretical probability.
   135→  1116→  const useLivePrior = liveN >= 20;
   136→  1117→  const livePriorFor = (gameName: string): number => {
   137→  1118→    if (!useLivePrior) return THEORETICAL[gameName] ?? 0.1;
   138→  1119→    return liveN > 0 ? (liveFreqAll.get(gameName) ?? 0) / liveN : (THEORETICAL[gameName] ?? 0.1);
   139→  1120→  };
   140→  1121→
   141→  1122→  // ===== RECENT LIVE SPIN FREQUENCY (last 10 real spins) =====
   142→  1123→  const liveRecent = liveSpins.slice(0, 10); // newest first in API response
   143→  1124→  const liveRecentN = liveRecent.length;
   144→  1125→  const liveRecentFreq = new Map<string, number>();
   145→  1126→  for (const g of GAMES) liveRecentFreq.set(g.name, 0);
   146→  1127→  for (const s of liveRecent) {
   147→  1128→    const gameName = SPIN_TO_GAME_NAME[s.sector] ?? s.sector;
   148→  1129→    if (liveRecentFreq.has(gameName)) {
   149→  1130→      liveRecentFreq.set(gameName, (liveRecentFreq.get(gameName) ?? 0) + 1);
   150→  1131→    }
   151→  1132→  }
   152→  1133→
   153→  1134→  // Per-game frequency in recent 10 USER-VERIFIED rounds (short-term)
   154→  1135→  const recentHist = hist.slice(-10);
   155→  1136→  const recentFreq = new Map<string, number>();
   156→  1137→  for (const g of GAMES) recentFreq.set(g.name, 0);
   157→  1138→  for (const h of recentHist) recentFreq.set(h.name, (recentFreq.get(h.name) ?? 0) + 1);
   158→  1139→
   159→  1140→  // Gap analysis per game
   160→  1141→  const gaps: Record<string, number> = {};
   161→  1142→  const gapHistories: Record<string, number[]> = {};
   162→  1143→  for (const g of GAMES) { gaps[g.name] = n; gapHistories[g.name] = []; }
   163→  1144→  let lastHitIdx: Record<string, number> = {};
   164→  1145→  for (const g of GAMES) lastHitIdx[g.name] = -1;
   165→  1146→  for (let i = 0; i < n; i++) {
   166→  1147→    const name = hist[i].name;
   167→  1148→    if (lastHitIdx[name] >= 0) {
   168→  1149→      gapHistories[name].push(i - lastHitIdx[name]);
   169→  1150→    }
   170→  1151→    lastHitIdx[name] = i;
   171→  1152→  }
   172→  1153→  for (const g of GAMES) {
   173→  1154→    gaps[g.name] = lastHitIdx[g.name] >= 0 ? n - 1 - lastHitIdx[g.name] : n;
   174→  1155→  }
   175→  1156→
   176→  1157→  // Adaptive weighting between long-term and recent
   177→  1158→  const adaptW = dashboard.adaptiveWeight;
   178→  1159→  const longW = 1 - adaptW;
   179→  1160→
   180→  1161→  // Per-game stability (variance of recent gaps → inverse)
   181→  1162→  const recentUnique = new Set(recentHist.map((g) => g.name)).size;
   182→  1163→  const stability = recentUnique >= 4 ? "STABLE" : recentUnique >= 2 ? "VOLATILE" : "STREAK";
   183→  1164→
   184→  1165→  // ===== REPEAT-PATTERN ANALYSIS (data-based, NOT blind carryover) =====
   185→  1166→  // Analyze the historical tendency of each outcome to REPEAT in the next
   186→  1167→  // round. This is NOT "last hit → predict same again". Instead, we measure:
   187→  1168→  //   P(next = X | current = X)  from ALL historical consecutive pairs.
   188→  1169→  // If the last actual result was X, and X has a historically HIGH repeat
   189→  1170→  // rate, then X gets a MILD evidence boost (capped). If X has a LOW repeat
   190→  1171→  // rate, no penalty — we just don't boost. This is pure statistical
   191→  1172→  // evidence, not a "last hit shortcut".
   192→  1173→  //
   193→  1174→  // We use BOTH the user-verified round history AND the live casino spins
   194→  1175→  // (mapped to game names) for a larger, more reliable sample.
   195→  1176→  const lastActualName = n > 0 ? hist[n - 1].name : null;
   196→  1177→  const lastLiveSpinName = liveN > 0
   197→  1178→    ? (SPIN_TO_GAME_NAME[liveSpins[0].sector] ?? null)
   198→  1179→    : null;
   199→  1180→  // Prefer the most recent data point (live spins are newer than user rounds).
   200→  1181→  const lastResultName = lastLiveSpinName ?? lastActualName;
   201→  1182→
   202→  1183→  // Build a combined consecutive-pair sequence from both sources.
   203→  1184→  // Live spins are newest-first; reverse to oldest-first for pair analysis.
   204→  1185→  const liveGameSeq = liveSpins
   205→  1186→    .map((s) => SPIN_TO_GAME_NAME[s.sector] ?? null)
   206→  1187→    .filter((x): x is string => x !== null)
   207→  1188→    .reverse(); // oldest-first
   208→  1189→  const userGameSeq = hist.map((h) => h.name);
   209→  1190→  const combinedSeq = [...userGameSeq, ...liveGameSeq];
   210→  1191→
   211→  1192→  // Count repeat pairs: P(next = X | current = X)
   212→  1193→  const repeatStats: Record<string, { currentCount: number; repeatCount: number; rate: number }> = {};
   213→  1194→  for (const g of GAMES) repeatStats[g.name] = { currentCount: 0, repeatCount: 0, rate: 0 };
   214→  1195→  for (let i = 0; i < combinedSeq.length - 1; i++) {
   215→  1196→    const cur = combinedSeq[i];
   216→  1197→    const next = combinedSeq[i + 1];
   217→  1198→    if (repeatStats[cur]) {
   218→  1199→      repeatStats[cur].currentCount++;
   219→  1200→      if (next === cur) repeatStats[cur].repeatCount++;
   220→  1201→    }
   221→  1202→  }
   222→  1203→  for (const g of GAMES) {
   223→  1204→    const rs = repeatStats[g.name];
   224→  1205→    rs.rate = rs.currentCount >= 3 ? rs.repeatCount / rs.currentCount : 0;
   225→  1206→  }
   226→  1207→  // Overall average repeat rate (for comparison baseline).
   227→  1208→  const totalRepeatCur = Object.values(repeatStats).reduce((s, r) => s + r.currentCount, 0);
   228→  1209→  const totalRepeatCount = Object.values(repeatStats).reduce((s, r) => s + r.repeatCount, 0);
   229→  1210→  const baselineRepeatRate = totalRepeatCur >= 5 ? totalRepeatCount / totalRepeatCur : 0;
   230→  1211→
   231→  1212→  // ===== BONUS CLUSTER DETECTION (bonus-aware scoring) =====
   232→  1213→  // Analyze whether bonus outcomes are: increasing, decreasing, clustering,
   233→  1214→  // appearing after specific sequences, repeating, or showing unusual recent
   234→  1215→  // frequency. This is NOT "bonus appeared → predict bonus next". Instead,
   235→  1216→  // we measure the CURRENT bonus activity level vs the long-term baseline.
   236→  1217→  // If bonus activity is elevated, bonus outcomes get a mild evidence boost
   237→  1218→  // (capped). This ensures the engine does NOT have a bonus blind spot.
   238→  1219→  const isBonusGame = (name: string): boolean => BONUS_NAMES.includes(name);
   239→  1220→
   240→  1221→  // Bonus frequency in combined sequence (long-term)
   241→  1222→  const combinedBonusCount = combinedSeq.filter(isBonusGame).length;
   242→  1223→  const combinedBonusRate = combinedSeq.length > 0 ? combinedBonusCount / combinedSeq.length : 0;
   243→  1224→  // Bonus frequency in recent 10 of combined sequence (short-term)
   244→  1225→  const recentCombined = combinedSeq.slice(-10);
   245→  1226→  const recentBonusCount = recentCombined.filter(isBonusGame).length;
   246→  1227→  const recentBonusRate = recentCombined.length > 0 ? recentBonusCount / recentCombined.length : 0;
   247→  1228→  // Bonus activity trend: recent vs long-term
   248→  1229→  const bonusTrend = recentBonusRate - combinedBonusRate; // positive = increasing
   249→  1230→  // Bonus clustering: count bonus "bursts" (2+ bonuses within 3 spins)
   250→  1231→  let bonusBursts = 0;
   251→  1232→  for (let i = 0; i < combinedSeq.length - 2; i++) {
   252→  1233→    const window = combinedSeq.slice(i, i + 3);
   253→  1234→    const bonusInWindow = window.filter(isBonusGame).length;
   254→  1235→    if (bonusInWindow >= 2) bonusBursts++;
   255→  1236→  }
   256→  1237→  // Is a bonus currently "active" (appeared in last 3 spins)?
   257→  1238→  const last3 = combinedSeq.slice(-3);
   258→  1239→  const bonusInLast3 = last3.filter(isBonusGame).length;
   259→  1240→  const bonusActive = bonusInLast3 > 0;
   260→  1241→  // Per-bonus-game recent frequency (last 10 combined)
   261→  1242→  const bonusRecentFreq: Record<string, number> = {};
   262→  1243→  for (const b of BONUS_NAMES) bonusRecentFreq[b] = 0;
   263→  1244→  for (const name of recentCombined) {
   264→  1245→    if (isBonusGame(name)) bonusRecentFreq[name] = (bonusRecentFreq[name] ?? 0) + 1;
   265→  1246→  }
   266→  1247→
   267→  1248→  const scores: CandidateScore[] = [];
   268→  1249→
   269→  1250→  for (let i = 0; i < GAMES.length; i++) {
   270→  1251→    const g = GAMES[i];
   271→  1252→    const theo = THEORETICAL[g.name] ?? 0.1;
   272→  1253→    // LONG-TERM PRIOR: prefer REAL casino spin frequency when available (20+
   273→  1254→    // real spins), else theoretical probability, else tiny user history.
   274→  1255→    const livePrior = livePriorFor(g.name);
   275→  1256→    const userLongFreq = n > 0 ? (freqAll.get(g.name) ?? 0) / n : 0;
   276→  1257→    // Blend user-verified long-term with live-prior when both exist.
   277→  1258→    // Live prior dominates (real casino data) but user history adds weight.
   278→  1259→    const longFreq = useLivePrior
   279→  1260→      ? livePrior * 0.7 + userLongFreq * 0.3
   280→  1261→      : userLongFreq;
   281→  1262→    // RECENT frequency: prefer live recent (last 10 real spins) when available;
   282→  1263→    // it's far more responsive than the tiny user recent slice.
   283→  1264→    const liveRecFreq = liveRecentN > 0 ? (liveRecentFreq.get(g.name) ?? 0) / liveRecentN : 0;
   284→  1265→    const userRecFreq = recentHist.length > 0 ? (recentFreq.get(g.name) ?? 0) / recentHist.length : 0;
   285→  1266→    const recFreq = useLivePrior
   286→  1267→      ? liveRecFreq * 0.6 + userRecFreq * 0.4
   287→  1268→      : userRecFreq;
   288→  1269→    const trend = recFreq - longFreq;
   289→  1270→    const gap = gaps[g.name];
   290→  1271→    const gapHistory = gapHistories[g.name];
   291→  1272→    const avgGap = gapHistory.length > 0
   292→  1273→      ? gapHistory.reduce((s, x) => s + x, 0) / gapHistory.length
   293→  1274→      : n / Math.max(freqAll.get(g.name) ?? 1, 1);
   294→  1275→    const isOverdue = avgGap > 0 && gap > avgGap * 1.5;
   295→  1276→    const isHot = longFreq > theo * 1.3 && (n >= 5 || liveN >= 20);
   296→  1277→    const isCold = longFreq < theo * 0.5 && (n >= 5 || liveN >= 20);
   297→  1278→
   298→  1279→    // Signal-wise historical performance (when this game was predicted)
   299→  1280→    const sw = dashboard.signalWiseHitRate[g.name];
   300→  1281→    const sampleSize = sw.predicted;
   301→  1282→    const signalHitRate = sw.rate;
   302→  1283→    const wilsonLower = sw.wilsonLower; // sample-size-aware confidence
   303→  1284→
   304→  1285→    // Volatility: variance of gap history
   305→  1286→    let volatility = 0;
   306→  1287→    if (gapHistory.length >= 2) {
   307→  1288→      const mean = gapHistory.reduce((s, x) => s + x, 0) / gapHistory.length;
   308→  1289→      volatility = gapHistory.reduce((s, x) => s + (x - mean) ** 2, 0) / gapHistory.length;
   309→  1290→    }
   310→  1291→
   311→  1292→    const signals: string[] = [];
   312→  1293→
   313→  1294→    // ===== SAMPLE-SIZE STABILIZATION (Bayesian / Laplace shrinkage) =====
   314→  1295→    // CRITICAL: Small-sample observations must NOT create extreme deviations.
   315→  1296→    //
   316→  1297→    // k=30 (increased from 20): rare outcomes need MORE evidence before their
   317→  1298→    // deviation can significantly affect the score. At k=30:
   318→  1299→    //   N=30, CRAZY TIME count=1, prior=1.85%:
   319→  1300→    //     smoothed = (1 + 30*0.0185) / 60 = 1.09% → dev=-41% (was +48% at k=20)
   320→  1301→    //   N=30, "1" count=12, prior=38.89%:
   321→  1302→    //     smoothed = (12 + 30*0.389) / 60 = 39.45% → dev=+1.4% (barely changed)
   322→  1303→    //
   323→  1304→    // This ensures rare outcomes need substantially more observations before
   324→  1305→    // they can displace high-prior common outcomes like "1" (38.89%) or "2" (24.07%).
   325→  1306→    const SHRINKAGE_K = 30;
   326→  1307→
   327→  1308→    // Compute observed count and total N for this outcome.
   328→  1309→    // Use the larger of (user rounds, live spins) as the sample.
   329→  1310→    const userCount = freqAll.get(g.name) ?? 0;
   330→  1311→    const liveCount = liveFreqAll.get(g.name) ?? 0;
   331→  1312→    // Total sample N = combined user rounds + live spins
   332→  1313→    const sampleN = n + liveN;
   333→  1314→    // Combined observed count (user + live, weighted by their respective N)
   334→  1315→    const combinedCount = userCount + liveCount;
   335→  1316→
   336→  1317→    // Laplace-smoothed frequency
   337→  1318→    const smoothedFreq = sampleN > 0
   338→  1319→      ? (combinedCount + SHRINKAGE_K * theo) / (sampleN + SHRINKAGE_K)
   339→  1320→      : theo;
   340→  1321→
   341→  1322→    // Raw (unstabilized) deviation — for debug display only
   342→  1323→    const rawBlendedFreq = (longFreq * longW + recFreq * adaptW);
   343→  1324→    const rawDeviation = theo > 0 ? (rawBlendedFreq - theo) / theo : 0;
   344→  1325→
   345→  1326→    // Stabilized deviation — uses the Laplace-smoothed frequency
   346→  1327→    const stabilizedDeviation = theo > 0 ? (smoothedFreq - theo) / theo : 0;
   347→  1328→
   348→  1329→    // Cap stabilized deviation to prevent extreme swings: [-0.6, +2.0]
   349→  1330→    const cappedDeviation = Math.max(-0.6, Math.min(2.0, stabilizedDeviation));
   350→  1331→
   351→  1332→    // ===== EXPERIMENTAL: RARE-OUTCOME EVIDENCE RELIABILITY LAYER =====
   352→  1333→    // (ACTIVE only in mode === "experimental"; baseline passes through unchanged)
   353→  1334→    //
   354→  1335→    // A small sample of rare-outcome appearances (e.g. PACHINKO 3× in 50
   355→  1336→    // rounds) can generate a large RELATIVE deviation (+39%) that, via
   356→  1337→    // evidenceScore = 1 + dev, displaces a substantially higher-prior
   357→  1338→    // number outcome. This is statistically unsound: the relative
   358→  1339→    // deviation of a rare outcome has much higher variance per observation
   359→  1340→    // than a common outcome, so the same count implies far less confidence.
