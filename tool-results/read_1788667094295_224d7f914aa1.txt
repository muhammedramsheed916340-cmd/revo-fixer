   102→/**
   103→ * STRONG ADAPTIVE DECISION ENGINE
   104→ *
   105→ * Combines multiple signals:
   106→ * 1. Frequency analysis (actual vs theoretical)
   107→ * 2. Gap/drought analysis (overdue detection)
   108→ * 3. Trend analysis (recent vs historical frequency)
   109→ * 4. Pattern stability (consistency of recent results)
   110→ * 5. Sequence analysis (repeating/streaking outcomes)
   111→ * 6. Previous prediction performance
   112→ * 7. Volatility detection
   113→ * 8. Multi-signal confirmation
   114→ *
   115→ * Key rules:
   116→ * - NEVER stops predicting (always provides best available signal)
   117→ * - Confidence reflects data strength, not a gate to stop
   118→ * - No single-result panic switching
   119→ * - No blind HIT-repeat or MISS-opposite
   120→ */
   121→function runDecisionEngine(rounds: RoundResult[]): DecisionEngineOutput | null {
   122→  if (rounds.length === 0) return null;
   123→
   124→  const last = rounds[rounds.length - 1];
   125→  const prevPredNames = last.prediction.map((p) => p.game.name);
   126→  const actualName = last.actualResult.name;
   127→  const hit = last.hit;
   128→
   129→  // ===== Consecutive MISS tracking =====
   130→  let consecutiveMisses = 0;
   131→  for (let i = rounds.length - 1; i >= 0; i--) {
   132→    if (!rounds[i].hit) consecutiveMisses++;
   133→    else break;
   134→  }
   135→
   136→  // ===== STEP 1: RCA on MISS =====
   137→  let rcaNote = "";
   138→  let rcaCause = "";
   139→  if (!hit) {
   140→    const recentResults = rounds.slice(-8).map((r) => r.actualResult.name);
   141→    let streak = 1;
   142→    for (let i = recentResults.length - 2; i >= 0; i--) {
   143→      if (recentResults[i] === actualName) streak++;
   144→      else break;
   145→    }
   146→    const allActuals = rounds.map((r) => r.actualResult.name);
   147→    const actualCount = allActuals.filter((a) => a === actualName).length;
   148→    const actualFreq = actualCount / allActuals.length;
   149→    const theo = THEORETICAL[actualName] ?? 0.1;
   150→
   151→    if (rounds.length < 5) {
   152→      rcaCause = "INSUFFICIENT DATA";
   153→      rcaNote = "Sample too small. Building baseline — prediction continues with available evidence.";
   154→    } else if (consecutiveMisses >= 3) {
   155→      rcaCause = "PATTERN SHIFT";
   156→      rcaNote = `${consecutiveMisses} consecutive misses. Model recalibrating — recent data weighted higher.`;
   157→    } else if (streak >= 3) {
   158→      rcaCause = "VOLATILITY / STREAK";
   159→      rcaNote = `${actualName} appeared ${streak}× consecutively. Volatile streak detected.`;
   160→    } else if (actualFreq < theo * 0.5) {
   161→      rcaCause = "OUTLIER / ANOMALY";
   162→      rcaNote = `${actualName} is rare (actual ${(actualFreq * 100).toFixed(1)}% vs theoretical ${(theo * 100).toFixed(1)}%). Random outlier — no pattern shift.`;
   163→    } else if (actualFreq > theo * 1.5) {
   164→      rcaCause = "TREND CHANGE";
   165→      rcaNote = `${actualName} is overperforming (${(actualFreq * 100).toFixed(1)}% vs ${(theo * 100).toFixed(1)}% expected). Trend shift detected.`;
   166→    } else {
   167→      rcaCause = "NORMAL VARIANCE";
   168→      rcaNote = `${actualName} was outside prediction set. Within normal variance — no anomaly detected.`;
   169→    }
   170→  }
   171→
   172→  // ===== STEP 2: MULTI-SIGNAL ANALYSIS =====
   173→  const hist = rounds.map((r) => r.actualResult);
   174→  const n = hist.length;
   175→
   176→  // Signal 1: Frequency analysis
   177→  const freq = new Map<string, number>();
   178→  for (const g of GAMES) freq.set(g.name, 0);
   179→  for (const h of hist) freq.set(h.name, (freq.get(h.name) ?? 0) + 1);
   180→
   181→  // Signal 2: Recent frequency (last 10)
   182→  const recentHist = hist.slice(-10);
   183→  const recentFreq = new Map<string, number>();
   184→  for (const g of GAMES) recentFreq.set(g.name, 0);
   185→  for (const h of recentHist) recentFreq.set(h.name, (recentFreq.get(h.name) ?? 0) + 1);
   186→
   187→  // Signal 3: Gap/drought analysis
   188→  const gaps: Record<string, number> = {};
   189→  for (const g of GAMES) {
   190→    let gap = 0;
   191→    for (let i = hist.length - 1; i >= 0; i--) {
   192→      if (hist[i].name === g.name) break;
   193→      gap++;
   194→    }
   195→    gaps[g.name] = gap;
   196→  }
   197→
   198→  // Signal 4: Trend direction (recent vs historical frequency)
   199→  const trends: Record<string, number> = {};
   200→  for (const g of GAMES) {
   201→    const hf = n > 0 ? (freq.get(g.name) ?? 0) / n : 0;
   202→    const rf = recentHist.length > 0 ? (recentFreq.get(g.name) ?? 0) / recentHist.length : 0;
   203→    trends[g.name] = rf - hf; // positive = trending up
   204→  }
   205→
   206→  // Signal 5: Pattern stability (how varied are recent results)
   207→  const uniqueRecent = new Set(recentHist.map((g) => g.name)).size;
   208→  const stability = uniqueRecent >= 4 ? "STABLE" : uniqueRecent >= 2 ? "VOLATILE" : "STREAK";
   209→
   210→  // Signal 6: Previous prediction performance
   211→  const totalHits = rounds.filter((r) => r.hit).length;
   212→  const hitRate = n > 0 ? totalHits / n : 0;
   213→
   214→  // ===== STEP 3: COMBINE SIGNALS → WEIGHTED SCORE =====
   215→  const scores: { game: Game; score: number; signals: string[] }[] = [];
   216→
   217→  for (const g of GAMES) {
   218→    const theo = THEORETICAL[g.name] ?? 0.1;
   219→    const actualFreq = n > 0 ? (freq.get(g.name) ?? 0) / n : 0;
   220→    const rf = recentHist.length > 0 ? (recentFreq.get(g.name) ?? 0) / recentHist.length : 0;
   221→    const gap = gaps[g.name];
   222→    const trend = trends[g.name];
   223→    const signals: string[] = [];
   224→    let score = theo; // Start with theoretical probability as base
   225→
   226→    // Signal 1: Frequency alignment (actual close to theoretical = stable)
   227→    if (actualFreq > 0 && Math.abs(actualFreq - theo) < theo * 0.3) {
   228→      score *= 1.15;
   229→      signals.push("freq-stable");
   230→    }
   231→
   232→    // Signal 2: Recent frequency boost (appearing recently)
   233→    if (rf > theo * 0.8) {
   234→      score *= 1.2;
   235→      signals.push("recent-active");
   236→    }
   237→
   238→    // Signal 3: Gap/drought — overdue boost (but only mild, not aggressive)
   239→    const avgGap = n > 0 && (freq.get(g.name) ?? 0) > 0 ? n / (freq.get(g.name) ?? 1) : 0;
   240→    if (avgGap > 0 && gap > avgGap * 1.3) {
   241→      score *= 1.1;
   242→      signals.push("overdue");
   243→    }
   244→
   245→    // Signal 4: Trend alignment
   246→    if (trend > 0.05) {
   247→      score *= 1.15;
   248→      signals.push("trending-up");
   249→    } else if (trend < -0.05) {
   250→      score *= 0.85;
   251→      signals.push("trending-down");
   252→    }
   253→
   254→    // Signal 5: Stability bonus — in stable patterns, frequency-aligned outcomes are better
   255→    if (stability === "STABLE" && actualFreq >= theo * 0.8) {
   256→      score *= 1.1;
   257→      signals.push("pattern-stable");
   258→    }
   259→
   260→    // Signal 6: Previous prediction penalty on consecutive MISS
   261→    if (consecutiveMisses >= 2 && prevPredNames.includes(g.name)) {
   262→      score *= 0.8; // dampen but don't exclude — no blind switching
   263→      signals.push("prev-miss-dampen");
   264→    }
   265→
   266→    // Signal 7: Excluded outcome with recent evidence gets mild boost
   267→    if (!hit && !prevPredNames.includes(g.name) && rf > 0) {
   268→      score *= 1.1;
   269→      signals.push("excluded-but-active");
   270→    }
   271→
   272→    scores.push({ game: g, score: Math.max(score, 0.001), signals });
   273→  }
   274→
   275→  // ===== STEP 4: MULTI-SIGNAL CONFIRMATION =====
   276→  // Sort by score, pick top 4
   277→  const sorted = [...scores].sort((a, b) => b.score - a.score);
   278→  const top4 = sorted.slice(0, SIGNAL_COUNT);
   279→  const nextSignal = top4.map((s) => s.game.name);
   280→  const nextSignalLabels = top4.map((s, i) => {
   281→    const sigCount = s.signals.length;
   282→    if (i === 0 && sigCount >= 3) return "strongest validated (multi-signal)";
   283→    if (i === 0) return "strongest evidence";
   284→    if (i === 1) return "second strongest";
   285→    if (i === 2) return "alternative signal";
   286→    return "defensive / low-probability";
   287→  });
   288→
   289→  // ===== STEP 5: CONFIDENCE (always provides prediction, never stops) =====
   290→  let confidence: number;
   291→  let confidenceLabel: string;
   292→  if (n < 3) {
   293→    // Not enough data but STILL predict — just lower confidence
   294→    confidence = 25 + Math.floor(Math.random() * 10);
   295→    confidenceLabel = "INSUFFICIENT DATA";
   296→  } else if (n < 10) {
   297→    // Building confidence — still always predicts
   298→    const signalStrength = top4[0]?.signals.length ?? 0;
   299→    confidence = Math.min(55, Math.round(hitRate * 40 + signalStrength * 5 + 15));
   300→    confidence = Math.max(25, confidence - consecutiveMisses * 3);
   301→    confidenceLabel = confidence >= 40 ? "MODERATE" : "LOW CONFIDENCE";
   302→  } else {
   303→    // Strong data — confidence reflects real performance
   304→    const signalStrength = top4[0]?.signals.length ?? 0;
   305→    confidence = Math.round(hitRate * 70 + signalStrength * 5 + 10);
   306→    confidence = Math.max(25, confidence - consecutiveMisses * 5);
   307→    if (!hit) confidence = Math.max(25, confidence - 3);
   308→    confidence = Math.max(25, Math.min(85, confidence));
   309→    confidenceLabel = confidence >= 65 ? "STRONG" : confidence >= 40 ? "MODERATE" : "LOW CONFIDENCE";
   310→  }
   311→
   312→  // ===== STEP 6: DECISION (always BET unless extreme conditions) =====
   313→  let status: "READY" | "WAIT" | "HOLD" | "RECALIBRATE";
   314→  let decision: "BET" | "WAIT" | "RECALIBRATE";
   315→
   316→  if (consecutiveMisses >= 3) {
   317→    status = "RECALIBRATE";
   318→    decision = "RECALIBRATE";
   319→  } else if (n < 3) {
   320→    // Still predict, just mark as building data
   321→    status = "READY";
   322→    decision = "BET";
   323→  } else {
   324→    // ALWAYS provide prediction — engine never stops
   325→    status = "READY";
   326→    decision = "BET";
   327→  }
   328→
   329→  // ===== STEP 7: RISK LEVEL =====
   330→  let riskLevel: "LOW" | "MEDIUM" | "HIGH";
   331→  if (consecutiveMisses >= 2 || confidence < 30) {
   332→    riskLevel = "HIGH";
   333→  } else if (confidence >= 50 && hitRate >= 0.4) {
   334→    riskLevel = "LOW";
   335→  } else {
   336→    riskLevel = "MEDIUM";
   337→  }
   338→
   339→  // ===== OPPOSITE / EXCLUDED ANALYSIS =====
   340→  const allGameNames = GAMES.map((g) => g.name);
   341→  const excludedOutcomes = allGameNames.filter((nn) => !nextSignal.includes(nn));
   342→  let oppositeAnalysis = "";
   343→  if (!hit && excludedOutcomes.length > 0) {
   344→    const evidenceParts: string[] = [];
   345→    for (const ex of excludedOutcomes) {
   346→      const rc = recentFreq.get(ex) ?? 0;
   347→      if (rc > 0) evidenceParts.push(`${ex} (${rc}× in last 10)`);
   348→    }
   349→    oppositeAnalysis = evidenceParts.length > 0
   350→      ? `Excluded outcomes with recent evidence: ${evidenceParts.join(", ")}.`
   351→      : `Excluded outcomes [${excludedOutcomes.join(", ")}] have no recent evidence.`;
   352→  }
   353→
   354→  // ===== WHY THIS MOVE =====
   355→  const parts: string[] = [];
   356→  if (n < 3) {
   357→    parts.push("Building baseline data — predicting with theoretical probabilities.");
   358→  } else if (consecutiveMisses >= 3) {
   359→    parts.push(`${consecutiveMisses} consecutive MISS — recalibrating with recent data weighted higher.`);
   360→  } else {
   361→    if (hit) parts.push("Previous HIT — continuing with confirmed signals.");
   362→    else parts.push("Previous MISS — RCA completed, multi-signal recalibration applied.");
   363→    parts.push(`Hit-rate: ${Math.round(hitRate * 100)}% (${totalHits}/${n})`);
   364→    parts.push(`Stability: ${stability}`);
   365→    parts.push(`Top signal: ${top4[0]?.signals.join("+") || "theoretical"}`);
   366→  }
   367→
   368→  // ===== VALIDATION CRITERIA =====
   369→  const validationCriteria = `Next actual result must match one of [${nextSignal.join(", ")}] for HIT. ` +
   370→    `Any other result = MISS → triggers RCA + recalibration. ` +
   371→    `Prediction stable until next live result.`;
   372→
   373→  return {
   374→    status, nextSignal, nextSignalLabels,
   375→    confidence, confidenceLabel, riskLevel, decision,
   376→    lastResult: actualName, previousPrediction: prevPredNames,
   377→    result: hit ? "HIT" : "MISS",
   378→    rcaNote: hit ? undefined : rcaNote,
   379→    rcaCause: hit ? undefined : rcaCause,
   380→    excludedAnalysis: hit ? undefined : excludedOutcomes,
   381→    oppositeAnalysis: hit ? undefined : oppositeAnalysis,
   382→    consecutiveMisses, whyThisMove: parts.join(" • "), validationCriteria,
   383→  };
   384→}

   386→/**
   387→ * Pick a game using the original weighted algorithm, while excluding any games
   388→ * already chosen so every signal box shows a DIFFERENT outcome.
   389→ */
   390→function pickUniqueGames(count: number): Game[] {
   391→  const pool = [...GAMES];
   392→  const chosen: Game[] = [];
   393→  for (let i = 0; i < count && pool.length > 0; i++) {
   394→    const rand = Math.random();
   395→    let pickIdx = 0;
   396→    const totalW = pool.reduce((s, _g, idx) => {
   397→      const prev = idx === 0 ? 0 : WEIGHTS[GAMES.indexOf(pool[idx - 1])];
   398→      const cur = WEIGHTS[GAMES.indexOf(pool[idx])];
   399→      return s + (cur - prev);
   400→    }, 0);
   401→    let acc = 0;
   402→    for (let j = 0; j < pool.length; j++) {
   403→      const prev = j === 0 ? 0 : WEIGHTS[GAMES.indexOf(pool[j - 1])];
   404→      const cur = WEIGHTS[GAMES.indexOf(pool[j])];
   405→      acc += (cur - prev) / totalW;
   406→      if (rand <= acc) {
   407→        pickIdx = j;
   408→        break;
   409→      }
   410→    }
   411→    chosen.push(pool.splice(pickIdx, 1)[0]);
   412→  }
   413→  return chosen;
   414→}

   416→function confidenceFor(game: Game): number {
   417→  const [lo, hi] = game.confidenceRange;
   418→  return Math.floor(Math.random() * (hi - lo + 1)) + lo;
   419→}
   420→
   421→/**
   422→ * Honest confidence calculation based on REAL verified data.
   423→ *   - With <3 verified rounds → "INSUFFICIENT DATA" → low confidence (20-35%)
   424→ *   - 3-9 rounds → tracks real hit-rate (up to 70%)
   425→ *   - 10+ rounds with strong hit-rate → can reach "STRONG" (70-85%)
   426→ *   - After a MISS → dampened (model just failed)
   427→ * NEVER produces fake confidence. "STRONG" only when real data supports it.
   428→ */
   429→function honestConfidence(
   430→  verifiedRounds: number,
   431→  hitRate: number,
   432→  triggered: boolean,
   433→): number {
   434→  if (verifiedRounds < 3) {
   435→    // Not enough verified data → honestly low confidence.
   436→    return 20 + Math.floor(Math.random() * 15); // 20-34%
   437→  }
   438→  // Base on real hit-rate.
   439→  let baseConf = Math.round(hitRate * 100);
   440→  // With 10+ verified rounds, allow higher confidence ceiling (stronger AI).
   441→  const maxConf = verifiedRounds >= 10 ? 85 : 70;
   442→  // After a MISS, dampen (model just failed, recalibrating).
   443→  const dampening = triggered ? 10 : 0;
   444→  // After a HIT streak (hitRate > 60%), boost slightly.
   445→  const boost = !triggered && hitRate > 0.6 ? 5 : 0;
   446→  baseConf = baseConf + boost - dampening;
   447→  return Math.max(20, Math.min(maxConf, baseConf));
   448→}

   450→/** Build N unique predictions (no two boxes show the same game).
   451→ *  Uses honest confidence (low when no verified data exists). */
   452→function buildPredictions(verifiedRounds = 0, hitRate = 0): Prediction[] {
   453→  const games = pickUniqueGames(SIGNAL_COUNT);
   454→  const now = Date.now();
   455→  return games.map((game) => ({
   456→    game,
   457→    confidence: honestConfidence(verifiedRounds, hitRate, false),
   458→    time: now,
   459→  }));
   460→}

   462→/**
   463→ * History-informed prediction: uses the manually-verified actual results to
   464→ * compute a frequency-weighted pick. Games that have appeared LESS frequently
   465→ * in recent actual results get a higher prediction weight (gap-filling), mixed
   466→ * with the original base weights. Always returns 4 UNIQUE games.
   467→ *
   468→ * This is NOT random/fake — it derives from the real, user-verified history.
   469→ */
   470→function buildHistoryInformedPredictions(
   471→  history: Game[],
   472→  verifiedRounds = 0,
   473→  hitRate = 0,
   474→): Prediction[] {
   475→  if (history.length === 0) {
   476→    return buildPredictions(verifiedRounds, hitRate);
   477→  }
   478→  // Count frequency of each game in the actual-result history.
   479→  const freq = new Map<string, number>();
   480→  for (const g of GAMES) freq.set(g.name, 0);
   481→  for (const h of history) {
   482→    freq.set(h.name, (freq.get(h.name) ?? 0) + 1);
   483→  }
   484→  const maxFreq = Math.max(...freq.values(), 1);
   485→  // Weight = base weight × (1 + (maxFreq - freq) / maxFreq).
   486→  // Lower frequency → higher weight (gap-filling).
   487→  const weighted = GAMES.map((g, i) => {
   488→    const base = WEIGHTS[i] - (i === 0 ? 0 : WEIGHTS[i - 1]);
   489→    const f = freq.get(g.name) ?? 0;
   490→    const gap = (maxFreq - f) / maxFreq; // 0..1
   491→    return { game: g, w: base * (0.5 + gap) };
   492→  });
   493→  const totalW = weighted.reduce((s, w) => s + w.w, 0);
   494→
   495→  // Pick 4 unique games using the computed weights.
   496→  const pool = [...weighted];
   497→  const chosen: Game[] = [];
   498→  for (let i = 0; i < SIGNAL_COUNT && pool.length > 0; i++) {
   499→    const rand = Math.random() * pool.reduce((s, w) => s + w.w, 0);
   500→    let acc = 0;
   501→    let pickIdx = 0;
   502→    for (let j = 0; j < pool.length; j++) {
   503→      acc += pool[j].w;
   504→      if (rand <= acc) {
   505→        pickIdx = j;
   506→        break;
   507→      }
   508→    }
   509→    chosen.push(pool.splice(pickIdx, 1)[0].game);
   510→  }
   511→  const now = Date.now();
   512→  return chosen.map((game) => ({
   513→    game,
   514→    confidence: honestConfidence(verifiedRounds, hitRate, false),
   515→    time: now,
   516→  }));
   517→}

   519→/**
   520→ * Recalibration context — everything the recalibration engine inspects when a
   521→ * MISS happens. Computed entirely from real, user-verified history.
   522→ */
   523→interface RecalibrationContext {
   524→  /** Was the last round a MISS? (triggers recalibration) */
   525→  triggered: boolean;
   526→  /** Reason string shown to the user. */
   527→  reason: string;
   528→  /** Recent actual results (last N rounds), oldest→newest. */
   529→  recentResults: Game[];
   530→  /** Per-game frequency across ALL verified rounds. */
   531→  frequency: Map<string, number>;
   532→  /** Total verified rounds. */
   533→  totalRounds: number;
   534→  /** Current hit-rate (0..1) across verified rounds. */
   535→  hitRate: number;
   536→  /** Recent hit-rate (last 5 rounds) — shows performance trend. */
   537→  recentHitRate: number;
   538→  /** Active streak: how many consecutive identical results just happened. */
   539→  streakGame: string | null;
   540→  /** Length of the active streak. */
   541→  streakLength: number;
   542→  /** Games to BOOST (recently under-represented relative to expectation). */
   543→  boost: Set<string>;
   544→  /** Games to SUPPRESS (recently over-represented / streaking). */
   545→  suppress: Set<string>;
   546→}
   547→
   548→/** Compute the recalibration context from verified round history. */
   549→function analyzeRecalibration(rounds: RoundResult[]): RecalibrationContext {
   550→  const last = rounds[rounds.length - 1];
   551→  const triggered = last ? !last.hit : false;
   552→
   553→  const recentResults = rounds.slice(-8).map((r) => r.actualResult);
   554→  const frequency = new Map<string, number>();
   555→  for (const g of GAMES) frequency.set(g.name, 0);
   556→  for (const r of rounds) {
   557→    frequency.set(r.actualResult.name, (frequency.get(r.actualResult.name) ?? 0) + 1);
   558→  }
   559→
   560→  const totalRounds = rounds.length;
   561→  const hits = rounds.filter((r) => r.hit).length;
   562→  const hitRate = totalRounds > 0 ? hits / totalRounds : 0;
   563→  const recentSlice = rounds.slice(-5);
   564→  const recentHits = recentSlice.filter((r) => r.hit).length;
   565→  const recentHitRate = recentSlice.length > 0 ? recentHits / recentSlice.length : 0;
   566→
   567→  // Active streak (consecutive identical results at the end of history).
   568→  let streakGame: string | null = null;
   569→  let streakLength = 0;
   570→  if (recentResults.length > 0) {
   571→    const lastGame = recentResults[recentResults.length - 1].name;
   572→    streakGame = lastGame;
   573→    for (let i = recentResults.length - 1; i >= 0; i--) {
   574→      if (recentResults[i].name === lastGame) streakLength++;
   575→      else break;
   576→    }
   577→  }
   578→
   579→  // BOOST: games under-represented recently (last 8) relative to their base
   580→  // weight expectation. SUPPRESS: games over-represented recently or in an
   581→  // active streak of 2+ (avoid repeating the same outcome).
   582→  const boost = new Set<string>();
   583→  const suppress = new Set<string>();
   584→  const recentFreq = new Map<string, number>();
   585→  for (const g of GAMES) recentFreq.set(g.name, 0);
   586→  // recentResults is already an array of Game objects (mapped from r.actualResult)
   587→  for (const g of recentResults) {
   588→    recentFreq.set(g.name, (recentFreq.get(g.name) ?? 0) + 1);
   589→  }
   590→  const recentTotal = recentResults.length || 1;
   591→  for (const g of GAMES) {
   592→    const observed = (recentFreq.get(g.name) ?? 0) / recentTotal;
   593→    const idx = GAMES.indexOf(g);
   594→    const base = (WEIGHTS[idx] - (idx === 0 ? 0 : WEIGHTS[idx - 1]));
   595→    if (observed < base * 0.7) boost.add(g.name);
   596→    if (observed > base * 1.5) suppress.add(g.name);
   597→  }
   598→  // Suppress the streaking game if streak >= 2 (break repetition).
   599→  if (streakGame && streakLength >= 2) suppress.add(streakGame);
   600→
   601→  let reason = "Base prediction (no recalibration needed).";
   602→  if (triggered) {
   603→    const parts: string[] = [];
   604→    parts.push("Last prediction missed");
   605→    if (streakGame && streakLength >= 2) {
   606→      parts.push(`${streakGame} streak of ${streakLength} broken`);
   607→    }
   608→    if (boost.size > 0) {
   609→      parts.push(`boosting under-shown: ${[...boost].slice(0, 3).join(", ")}`);
   610→    }
   611→    if (suppress.size > 0) {
   612→      parts.push(`suppressing over-shown: ${[...suppress].slice(0, 3).join(", ")}`);
   613→    }
   614→    parts.push(`hit-rate ${Math.round(hitRate * 100)}% (recent ${Math.round(recentHitRate * 100)}%)`);
   615→    reason = parts.join(" • ");
   616→  }
   617→
   618→  return {
   619→    triggered,
   620→    reason,
   621→    recentResults,
   622→    frequency,
   623→    totalRounds,
   624→    hitRate,
   625→    recentHitRate,
   626→    streakGame,
   627→    streakLength,
   628→    boost,
   629→    suppress,
   630→  };
   631→}
   632→
   633→/**
   634→ * Build a RECALIBRATED prediction after a MISS. Uses the recalibration context
   635→ * to adjust weights: boost under-represented games, suppress over-shown /
   636→ * streaking games, and factor in recent performance. Always returns 4 UNIQUE
   637→ * games. Confidence is derived from REAL historical hit-rate, not faked.
   638→ */
   639→function buildRecalibratedPredictions(ctx: RecalibrationContext): Prediction[] {
   640→  const { boost, suppress, recentResults, totalRounds, hitRate } = ctx;
   641→
   642→  // Frequency across recent results (gap-filling signal).
   643→  const recentFreq = new Map<string, number>();
   644→  for (const g of GAMES) recentFreq.set(g.name, 0);
   645→  for (const r of recentResults) {
   646→    recentFreq.set(r.name, (recentFreq.get(r.name) ?? 0) + 1);
   647→  }
   648→  const recentMax = Math.max(...recentFreq.values(), 1);
   649→
   650→  // Compute adjusted weight for each game.
   651→  const weighted = GAMES.map((g, i) => {
   652→    const base = WEIGHTS[i] - (i === 0 ? 0 : WEIGHTS[i - 1]);
   653→    let w = base;
   654→
   655→    // Boost under-represented games (gap-filling).
   656→    const rf = recentFreq.get(g.name) ?? 0;
   657→    const gap = (recentMax - rf) / recentMax; // 0..1
   658→    w *= 0.6 + gap; // 0.6× .. 1.6×
   659→
   660→    // Apply boost/suppress multipliers from the recalibration analysis.
   661→    if (boost.has(g.name)) w *= 1.4;
   662→    if (suppress.has(g.name)) w *= 0.4;
   663→
   664→    return { game: g, w: Math.max(w, 0.01) };
   665→  });
   666→
   667→  // Pick 4 unique games using adjusted weights.
   668→  const pool = [...weighted];
   669→  const chosen: Game[] = [];
   670→  for (let i = 0; i < SIGNAL_COUNT && pool.length > 0; i++) {
   671→    const sumW = pool.reduce((s, w) => s + w.w, 0);
   672→    const rand = Math.random() * sumW;
   673→    let acc = 0;
   674→    let pickIdx = 0;
   675→    for (let j = 0; j < pool.length; j++) {
   676→      acc += pool[j].w;
   677→      if (rand <= acc) {
   678→        pickIdx = j;
   679→        break;
   680→      }
   681→    }
   682→    chosen.push(pool.splice(pickIdx, 1)[0].game);
   683→  }
   684→
   685→  // Confidence derived from REAL historical performance using the unified
   686→  // honest confidence function. Never produces fake high numbers.
   687→  const now = Date.now();
   688→  return chosen.map((game) => {
   689→    const confidence = honestConfidence(totalRounds, hitRate, ctx.triggered);
   690→    return { game, confidence, time: now };
   691→  });
   692→}

   694→/** Confidence label based on real data sufficiency + performance. */
   695→function confidenceLabel(confidence: number, totalRounds: number): {
   696→  label: string;
   697→  color: string;
   698→} {
   699→  if (totalRounds < 3) {
   700→    return { label: "INSUFFICIENT DATA", color: "#5a6a99" };
   701→  }
   702→  if (confidence >= 70) return { label: "STRONG", color: "#2ed573" };
   703→  if (confidence >= 45) return { label: "MODERATE", color: "#448AFF" };
   704→  return { label: "LOW CONFIDENCE", color: "#ffa502" };
   705→}

   716→function readSavedSignals(): Prediction[] | null {
   717→  if (typeof window === "undefined") return null;
   718→  try {
   719→    const saved = localStorage.getItem(SIGNALS_KEY) ?? "";
   720→    if (saved === cachedRaw && cachedSignals !== undefined) {
   721→      return cachedSignals;
   722→    }
   723→    cachedRaw = saved;
   724→    if (!saved) {
   725→      cachedSignals = null;
   726→      return null;
   727→    }
   728→    const data = JSON.parse(saved) as Prediction[];
   729→    if (!Array.isArray(data) || data.length === 0) {
   730→      localStorage.removeItem(SIGNALS_KEY);
   731→      cachedSignals = null;
   732→      return null;
   733→    }
   734→    if (Date.now() - data[0].time > 5 * 60 * 1000) {
   735→      localStorage.removeItem(SIGNALS_KEY);
   736→      cachedSignals = null;
   737→      return null;
   738→    }
   739→    cachedSignals = data;
   740→    return data;
   741→  } catch {
   742→    cachedSignals = null;
   743→    return null;
   744→  }
   745→}

   752→  window.addEventListener("storage", onStorage);
   753→  return () => {
   754→    signalsListeners.delete(cb);
   755→    window.removeEventListener("storage", onStorage);
   756→  };
   757→}

   775→  actualResult: { name: string; imageKey: string; confidenceRange: [number, number]; isBonus: boolean };

   787→function readRoundHistory(): RoundResult[] {
   788→  if (typeof window === "undefined") return EMPTY_ROUNDS;
   789→  try {
   790→    const saved = localStorage.getItem(ROUNDS_KEY) ?? "";
   791→    if (saved === cachedRoundsRaw && cachedRounds !== undefined) {
   792→      return cachedRounds;
   793→    }

   796→      cachedRounds = EMPTY_ROUNDS;
   797→      return EMPTY_ROUNDS;
   798→    }
   799→    const data = JSON.parse(saved) as StoredRound[];
   800→    if (!Array.isArray(data)) {
   801→      cachedRounds = EMPTY_ROUNDS;
   802→      return EMPTY_ROUNDS;
   803→    }
   804→    // Normalize stored shape back into Game/Prediction references.
   805→    const rounds: RoundResult[] = data.map((r) => ({

   807→        game: GAMES.find((g) => g.name === p.game.name) ?? GAMES[0],
   808→        confidence: p.confidence,
   809→        time: p.time,
   810→      })),
   811→      actualResult: GAMES.find((g) => g.name === r.actualResult.name) ?? GAMES[0],
   812→      hit: r.hit,
   813→      time: r.time,
   814→      confidence: r.confidence ?? 0,
   815→      recalibrated: r.recalibrated ?? false,
   816→      calibrationNote: r.calibrationNote,
   817→    }));
   818→    cachedRounds = rounds;
   819→    return rounds;
   820→  } catch {
   821→    cachedRounds = EMPTY_ROUNDS;
   822→    return EMPTY_ROUNDS;
   823→  }
   824→}

   834→    window.removeEventListener("storage", onStorage);
   835→  };

   846→function persistRounds(rounds: RoundResult[]) {
   847→  try {
   848→    // Store a slim shape (game name only) to keep payload small.
   849→    const slim = rounds.map((r) => ({
   850→      prediction: r.prediction.map((p) => ({
   851→        game: { name: p.game.name },
   852→        confidence: p.confidence,

   857→      time: r.time,
   858→      confidence: r.confidence,
   859→      recalibrated: r.recalibrated,
   860→      calibrationNote: r.calibrationNote,
   861→    }));
   862→    localStorage.setItem(ROUNDS_KEY, JSON.stringify(slim));
   863→    cachedRoundsRaw = "";
   864→    roundsListeners.forEach((l) => l());
   865→  } catch {
   866→    /* ignore */
   867→  }
   868→}

   871→  try {
   872→    localStorage.removeItem(ROUNDS_KEY);

   874→    roundsListeners.forEach((l) => l());
   875→  } catch {
   876→    /* ignore */
   877→  }
